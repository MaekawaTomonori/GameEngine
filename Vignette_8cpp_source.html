77 std::string path = "./Assets/Data/PostEffect/Vignette/" + _presetName + ".json";
78
79 // ファイル不在時はデフォルト値

86 }
87
88 std::ifstream file(path);
89 if (!file.is_open()) {
90 Log::Send(Log::Level::WARNING, std::format("Failed to open preset file: {}", path));

92 }

94 json paramJson;
95 file >> paramJson;
96 file.close();
97
98 // 任意名キーフレーム読み込み
99 keyframes_.clear();
100 for (auto& [name, data] : paramJson.items()) {
101 if (name == "keyframes") continue;
102 KeyframeData kf;
103 kf.intensity = data["intensity"];
104 kf.scale = data["scale"];
105 kf.color = Vector4(data["color"][0], data["color"][1], data["color"][2]);
106 keyframes_[name] = kf;
107 }
108
109 if (paramJson.contains("keyframes")) {
110 keyframeOrder_ = paramJson["keyframes"].get<std::vector<std::string>>();
111 }
112
113 // キーフレーム検証
114 for (const auto& name : keyframeOrder_) {
115 if (keyframes_.find(name) == keyframes_.end()) {
116 Log::Send(Log::Level::WARNING, std::format("Keyframe '{}' not _found, using defaults", name));
117 keyframes_.clear();
118 keyframes_["Start"] = {0.0f, 1.0f, Vector4(0, 0, 0, 1)};
119 keyframes_["End"] = {0.5f, 1.2f, Vector4(0.05f, 0.05f, 0.05f, 1.f)};
120 keyframeOrder_ = {"Start", "End"};
121 break;
122 }

126void Vignette::UpdateAnimation(float _t) {
127 // Q43: 単一キーフレーム対応
128 if (keyframeOrder_.size() == 1) {
129 const auto& kf = keyframes_[keyframeOrder_[0]];
130 material_->intensity = kf.intensity;
131 material_->scale = kf.scale;
132 material_->color = Vector4(kf.color.x, kf.color.y, kf.color.z, material_->color.w);
133 return;
134 }
135
136 if (keyframeOrder_.size() < 2) return;
137
138 // セグメント計算
139 float segmentCount = static_cast<float>(keyframeOrder_.size() - 1);
140 float segmentProgress = _t * segmentCount;
141 int currentSegment = static_cast<int>(segmentProgress);
142 float segmentT = segmentProgress - currentSegment;
143
144 if (currentSegment >= static_cast<int>(segmentCount)) {
145 currentSegment = static_cast<int>(segmentCount) - 1;
146 segmentT = 1.0f;
147 }
148
149 const auto& startKf = keyframes_[keyframeOrder_[currentSegment]];
150 const auto& endKf = keyframes_[keyframeOrder_[currentSegment + 1]];
151
152 // Linear補間
153 material_->intensity = std::lerp(startKf.intensity, endKf.intensity, segmentT);
154 material_->scale = std::lerp(startKf.scale, endKf.scale, segmentT);
155 Vector4 color = MathUtils::Lerp(startKf.color, endKf.color, segmentT);
156 material_->color = Vector4(color.x, color.y, color.z, material_->color.w);
157}
