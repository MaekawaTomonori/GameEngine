44void DrawVUBar(ImDrawList* dl, float x, float y, float w, float h, float level) {
45 dl->AddRectFilled({ x, y }, { x + w, y + h }, IM_COL32(20, 20, 20, 255), 2.f);
46 if (level <= 0.001f) return;
47
48 float fillH = h * std::min(level, 1.0f);
49 float greenH = h * 0.75f;
50 float yellowH = h * 0.15f;
51 float greenTop = y + h - greenH;
52 float yellowTop = greenTop - yellowH;
53
54 if (fillH <= greenH) {
55 dl->AddRectFilled({ x, y + h - fillH }, { x + w, y + h }, IM_COL32(40, 200, 60, 255), 2.f);
56 } else if (fillH <= greenH + yellowH) {
57 float yf = fillH - greenH;
58 dl->AddRectFilled({ x, greenTop }, { x + w, y + h }, IM_COL32(40, 200, 60, 255), 2.f);
59 dl->AddRectFilled({ x, greenTop - yf }, { x + w, greenTop }, IM_COL32(220, 200, 30, 255), 2.f);
60 } else {
61 float rf = fillH - greenH - yellowH;
62 dl->AddRectFilled({ x, greenTop }, { x + w, y + h }, IM_COL32(40, 200, 60, 255), 2.f);
63 dl->AddRectFilled({ x, yellowTop }, { x + w, greenTop }, IM_COL32(220, 200, 30, 255), 2.f);
64 dl->AddRectFilled({ x, yellowTop - rf }, { x + w, yellowTop}, IM_COL32(220, 60, 40, 255), 2.f);
65 }
66}

68float SimulateVU(int seed, float time) {
69 float f1 = fabsf(sinf(time * 8.5f + static_cast<float>(seed) * 2.1f));
70 float f2 = fabsf(sinf(time * 17.3f + static_cast<float>(seed) * 0.9f));
71 float f3 = fabsf(sinf(time * 33.7f + static_cast<float>(seed) * 1.5f));
72 return std::clamp(0.55f * f1 + 0.30f * f2 + 0.15f * f3, 0.0f, 1.0f);
73}

552void AudioDebugPanel::DrawPlaybackList(std::string_view name, ImVec4 headerColor,
553 std::vector<PlayingItem>& items) {
554 {
555 ImGui::PushStyleColor(ImGuiCol_Text, headerColor);
556 ImGui::TextUnformatted(name.data());
557 ImGui::PopStyleColor();
558 ImGui::SameLine();
559 ImGui::PushStyleColor(ImGuiCol_Text, DawStyle::kTextDim);

561 ImGui::PopStyleColor();
562 }
563 ImGui::Separator();
564
565 if (items.empty()) {
566 ImGui::PushStyleColor(ImGuiCol_Text, DawStyle::kTextDim);
567 ImGui::TextUnformatted("No active playbacks");
568 ImGui::PopStyleColor();
569 return;
570 }
571
572 constexpr ImGuiTableFlags kTableFlags =
573 ImGuiTableFlags_BordersInnerV | ImGuiTableFlags_RowBg | ImGuiTableFlags_ScrollY;
574 if (!ImGui::BeginTable("##items", 7, kTableFlags)) return;
575
576 ImGui::TableSetupScrollFreeze(0, 1);
577 ImGui::TableSetupColumn("Sound", ImGuiTableColumnFlags_WidthFixed, 82.f);
578 ImGui::TableSetupColumn("Vol", ImGuiTableColumnFlags_WidthFixed, 72.f);
579 ImGui::TableSetupColumn("Pan", ImGuiTableColumnFlags_WidthFixed, 72.f);
580 ImGui::TableSetupColumn("M", ImGuiTableColumnFlags_WidthFixed, 22.f);
581 ImGui::TableSetupColumn("L", ImGuiTableColumnFlags_WidthFixed, 22.f);
582 ImGui::TableSetupColumn("Trans", ImGuiTableColumnFlags_WidthFixed, 58.f);
583 ImGui::TableSetupColumn("##led", ImGuiTableColumnFlags_WidthFixed, 14.f);
584 ImGui::TableHeadersRow();
585
586 for (int i = 0; i < static_cast<int>(items.size()); ++i) {
587 PlayingItem& item = items[i];
588 const bool playing = item.playback.IsPlaying();
589
590 ImGui::PushID(i);
591 ImGui::TableNextRow();
592
593 // Sound name
594 ImGui::TableSetColumnIndex(0);
595 std::string lbl = item.soundName;
596 const size_t sl = lbl.find_last_of("/\\");
597 if (sl != std::string::npos) lbl = lbl.substr(sl + 1);
598 if (lbl.size() > 9) lbl = lbl.substr(0, 8) + "~";
599 ImGui::TextUnformatted(lbl.c_str());
600 if (ImGui::IsItemHovered()) ImGui::SetTooltip("%s", item.soundName.c_str());
601
602 // Volume
603 ImGui::TableSetColumnIndex(1);
604 ImGui::SetNextItemWidth(68.f);
605 if (ImGui::SliderFloat("##vol", &item.volume, 0.f, 1.f, "%.2f"))
606 item.playback.SetVolume(item.volume);
607
608 // Pan
609 ImGui::TableSetColumnIndex(2);
610 ImGui::SetNextItemWidth(68.f);
611 if (ImGui::SliderFloat("##pan", &item.pan, -1.f, 1.f, "%.2f"))
612 item.playback.SetPan(item.pan);
613
614 // Mute
615 ImGui::TableSetColumnIndex(3);
616 {
617 const bool wasMuted = item.muted;
618 if (wasMuted) {
619 ImGui::PushStyleColor(ImGuiCol_Button, DawStyle::kMuteActive);
620 ImGui::PushStyleColor(ImGuiCol_ButtonHovered, { 1.f, 0.4f, 0.4f, 1.f });
621 ImGui::PushStyleColor(ImGuiCol_ButtonActive, { 0.6f, 0.1f, 0.1f, 1.f });
622 }
623 if (ImGui::SmallButton("M")) { item.muted = !item.muted; item.playback.Mute(item.muted); }
624 if (ImGui::IsItemHovered()) ImGui::SetTooltip("Mute");
625 if (wasMuted) ImGui::PopStyleColor(3);
626 }
627
628 // Loop
629 ImGui::TableSetColumnIndex(4);
630 {
631 const bool wasLoop = item.loop;
632 if (wasLoop) {
633 ImGui::PushStyleColor(ImGuiCol_Button, DawStyle::kLoopActive);
634 ImGui::PushStyleColor(ImGuiCol_ButtonHovered, { 0.4f, 0.7f, 1.f, 1.f });
635 ImGui::PushStyleColor(ImGuiCol_ButtonActive, { 0.1f, 0.35f, 0.7f, 1.f });
636 }
637 if (ImGui::SmallButton("L")) { item.loop = !item.loop; item.playback.SetLoop(item.loop); }
638 if (ImGui::IsItemHovered()) ImGui::SetTooltip("Loop");
639 if (wasLoop) ImGui::PopStyleColor(3);
640 }
641
642 // Transport: Pause/Resume | Stop
643 ImGui::TableSetColumnIndex(5);
644 {
645 const bool wasPaused = item.paused;
646 if (wasPaused) {
647 ImGui::PushStyleColor(ImGuiCol_Button, DawStyle::kPauseActive);
648 ImGui::PushStyleColor(ImGuiCol_ButtonHovered, { 0.9f, 0.78f, 0.18f, 1.f });
649 ImGui::PushStyleColor(ImGuiCol_ButtonActive, { 0.5f, 0.40f, 0.05f, 1.f });
650 }
651 if (ImGui::SmallButton("||")) {
652 if (playing && !item.paused) { item.playback.Pause(); item.paused = true; }
653 else if (item.paused) { item.playback.Resume(); item.paused = false; }
654 }
655 if (ImGui::IsItemHovered()) ImGui::SetTooltip("Pause / Resume");
656 if (wasPaused) ImGui::PopStyleColor(3);
657
658 ImGui::SameLine(0, 3.f);
659
660 if (ImGui::SmallButton("[]")) { item.playback.Stop(); item.paused = false; }
661 if (ImGui::IsItemHovered()) ImGui::SetTooltip("Stop");
662 }
663
664 // Playback state LED
665 ImGui::TableSetColumnIndex(6);
666 {
667 ImDrawList* dl = ImGui::GetWindowDrawList();
668 ImVec2 pos = ImGui::GetCursorScreenPos();
669 float r = 5.f;
670 ImVec2 ctr = { pos.x + r, pos.y + r + 3.f };
671 ImU32 col = (playing && !item.paused) ? IM_COL32(50, 255, 80, 255)
672 : item.paused ? IM_COL32(220, 190, 30, 255)
673 : IM_COL32(28, 60, 28, 255);
674 dl->AddCircleFilled(ctr, r, col);
675 dl->AddCircle(ctr, r, IM_COL32(0, 0, 0, 160), 12, 1.f);
676 ImGui::Dummy({ r * 2.f, r * 2.f + 6.f });
677 }
678
679 ImGui::PopID();
680 }
681
682 ImGui::EndTable();
683}
