14void Emitter::Initialize(const MeshData& _mesh) {
15 if (!adapter_) {

25 uuid_ = Utils::GenerateUniqueId();

27 data_ = _mesh;

29 // Vertex Buffer
30 vr_ = adapter_->CreateBufferResource(sizeof(Vertex) * data_.vertices.size());
31 vbv_.BufferLocation = vr_->Get()->GetGPUVirtualAddress();
32 vbv_.SizeInBytes = static_cast<UINT>(sizeof(Vertex) * data_.vertices.size());
33 vbv_.StrideInBytes = sizeof(Vertex);
34 Vertex* vd = nullptr;
35 vr_->Get()->Map(0, nullptr, reinterpret_cast<void**>(&vd));
36 vd_ = {vd, data_.vertices.size()};
37 std::copy_n(data_.vertices.data(), data_.vertices.size(), vd_.data());
38
39 // Index Buffer
40 if (!data_.indices.empty()) {
41 ir_ = adapter_->CreateBufferResource(sizeof(uint32_t) * data_.indices.size());
42 ibv_.BufferLocation = ir_->Get()->GetGPUVirtualAddress();
43 ibv_.SizeInBytes = static_cast<UINT>(sizeof(uint32_t) * data_.indices.size());
44 ibv_.Format = DXGI_FORMAT_R32_UINT;
45 uint32_t* id = nullptr;
46 ir_->Get()->Map(0, nullptr, reinterpret_cast<void**>(&id));
47 id_ = { id, data_.indices.size() };
48 std::copy_n(data_.indices.data(), data_.indices.size(), id_.data());
49 }
50
51 // Mesh で言うMaterial
52 ForGpu* forMap = nullptr;
53 resource_ = adapter_->CreateBufferResource(sizeof(ForGpu) * MAX);
54 resource_->Get()->Map(0, nullptr, reinterpret_cast<void**>(&forMap));
55 mapped_ = {forMap, MAX};
56
57 index_ = srv_->Allocate();
58 handle_ = srv_->GetGPUHandle(index_);
59 srv_->CreateSRVForStructuredBuffer(index_, resource_->Get(), MAX, sizeof(ForGpu));
60
61 Singleton<TextureManager>::GetInstance()->Load(texture_);
62}

70void Emitter::Draw() {
71 if (!adapter_) return;
72 if (actives_ <= 0) return;
73
74 const auto command = adapter_->GetCommandList();
75
76 command->IASetPrimitiveTopology(D3D_PRIMITIVE_TOPOLOGY_TRIANGLELIST);
77 command->IASetVertexBuffers(0, 1, &vbv_);
78 command->SetGraphicsRootDescriptorTable(0, handle_); // Parameter 0: Particle Buffer
79 command->SetGraphicsRootDescriptorTable(1, Singleton<TextureManager>::GetInstance()->GetGPUHandle(texture_)); // Parameter 1: Texture
80
81 if (ir_) {
82 command->IASetIndexBuffer(&ibv_);
83 command->DrawIndexedInstanced(static_cast<UINT>(id_.size()), actives_, 0, 0, 0);
84 }
85 else {
86 command->DrawInstanced(static_cast<UINT>(vd_.size()), actives_, 0, 0);

262void Emitter::FrequencyUpdate() {
263 if (!active_) return;
264 if (frequency_ <= 0.01f) {
265 active_ = false;
266 return;
267 }
268
269 elapsedTime_ += Time::GetDeltaTime();
270 if (elapsedTime_ >= duration_) {
271 active_ = false;
272 return;
273 }
274
275 if (frequency_ <= timer_) {
276 timer_ = 0.f;
277 Spawn(spawnCount_);
278 return;
279 }
280
281 timer_ += Time::GetDeltaTime();
282}

284void Emitter::Spawn(const uint16_t& _count) {
285 for (uint16_t i = 0; i < _count; ++i) {
286 Vector3 spawnPos = position_;
287 Vector3 spawnVel = velocity_;
288 if (spawnFunc_) {
289 spawnFunc_(position_, spawnPos, spawnVel);
290 }
291
292 std::unique_ptr<Particle> particle = std::make_unique<Particle>();
293 particle->SetOrigin(position_)
294 .SetPosition(spawnPos)
295 .SetScale(size_)
296 .SetColor(color_)
297 .SetVelocity(spawnVel)
298 .SetRotation(rotation_)
299 .SetRotationVelocity(rotationVelocity_)
300 .SetColorKeys(colorKeys_)
301 .SetSizeKeys(sizeKeys_)
302 .SetUpdateFunction(updateFunc_)
303 .Initialize(particleLifetime_);
304 particles_.emplace_back(std::move(particle));
305 }
306}

308void Emitter::RegisterGpu() {
309 const auto& ac = Singleton<CameraController>::GetInstance()->GetActive();
310
311 Matrix4x4 rotation;
312 if (billboard_) {
313 rotation = BackToFront * ac->GetMatrix();
314 rotation.matrix[3][0] = 0.f;
315 rotation.matrix[3][1] = 0.f;
316 rotation.matrix[3][2] = 0.f;
317 } else {
318 rotation = MathUtils::Matrix::MakeIdentity();
319 }
320
321 actives_ = 0;
322
323 std::erase_if(particles_, [&](const auto& _p) { return _p->IsDead(); });
324
325 if (particles_.empty()) return;
326
327 for (auto& particle : particles_) {
328 particle->Update();
329
330 const Vector3 rot = particle->GetRotation();
331 const Matrix4x4 particleRot = MathUtils::Matrix::MakeRotateX(rot.x)
332 * MathUtils::Matrix::MakeRotateY(rot.y)
333 * MathUtils::Matrix::MakeRotateZ(rot.z);
334
335 mapped_[actives_].world = MathUtils::Matrix::MakeAffineMatrix(
336 MathUtils::Matrix::MakeScaleMatrix(particle->GetScale()),
337 particleRot * rotation,
338 MathUtils::Matrix::MakeTranslateMatrix(particle->GetPosition())
339 );
340 mapped_[actives_].wvp = mapped_[actives_].world * ac->GetViewProjection();
341 mapped_[actives_].color = particle->GetColor();
342 ++actives_;
343 }
