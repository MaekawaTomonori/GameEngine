199void Emitter::Spawn(const uint16_t& _count) {
200 for (uint16_t i = 0; i < _count; ++i) {
201 Vector3 spawnPos = position_;
202 Vector3 spawnVel = velocity_;
203 if (spawnFunc_) {
204 spawnFunc_(position_, spawnPos, spawnVel);
205 }
206
207 std::unique_ptr<Particle> particle = std::make_unique<Particle>();
208 particle->SetOrigin(position_)
209 .SetPosition(spawnPos)
210 .SetScale(size_)
211 .SetColor(color_)
212 .SetVelocity(spawnVel)
213 .SetUpdateFunction(updateFunc_)
214 .Initialize(3.f);
215 particles_.emplace_back(std::move(particle));
216 }

219void Emitter::RegisterGpu() {
220 const auto& ac = Singleton<CameraController>::GetInstance()->GetActive();
221 Matrix4x4 billboard = BackToFront * ac->GetMatrix();
222 billboard.matrix[3][0] = 0.f;
223 billboard.matrix[3][1] = 0.f;
224 billboard.matrix[3][2] = 0.f;
225 actives_ = 0;
226
227 std::erase_if(particles_, [&](const auto& _p) { return _p->IsDead(); });
228
229 if (particles_.empty())return;
230
231 for (auto& particle : particles_) {
232 particle->Update();
233 mapped_[actives_].world = MathUtils::Matrix::MakeAffineMatrix(
234 MathUtils::Matrix::MakeScaleMatrix(particle->GetScale()),
235 billboard,
236 MathUtils::Matrix::MakeTranslateMatrix(particle->GetPosition())
237 );
238 mapped_[actives_].wvp = mapped_[actives_].world * ac->GetViewProjection();
239 mapped_[actives_].color = particle->GetColor();
240 ++actives_;
241 }
