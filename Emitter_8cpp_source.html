31 vbv_.BufferLocation = vr_->Get()->GetGPUVirtualAddress();
32 vbv_.SizeInBytes = static_cast<UINT>(sizeof(Vertex) * data_.vertices.size());
33 vbv_.StrideInBytes = sizeof(Vertex);

42 ibv_.BufferLocation = ir_->Get()->GetGPUVirtualAddress();
43 ibv_.SizeInBytes = static_cast<UINT>(sizeof(uint32_t) * data_.indices.size());
44 ibv_.Format = DXGI_FORMAT_R32_UINT;
45 uint32_t* id = nullptr;
46 ir_->Get()->Map(0, nullptr, reinterpret_cast<void**>(&id));

48 std::copy_n(data_.indices.data(), data_.indices.size(), id_.data());
49 }

269void Emitter::FrequencyUpdate() {
270 if (!active_) return;
271 if (frequency_ <= 0.01f) {
272 active_ = false;

291void Emitter::Spawn(const uint16_t& _count) {
292 for (uint16_t i = 0; i < _count; ++i) {
293 if (actives_ >= MAX) return;
294
295 Vector3 spawnPos = position_;
296 Vector3 spawnVel = velocity_;
297 if (spawnFunc_) {
298 spawnFunc_(position_, spawnPos, spawnVel);
299 }
300
301 Particle& particle = particlePool_[actives_];
302 particle.SetOrigin(position_)
303 .SetPosition(spawnPos)
304 .SetScale(size_)
305 .SetColor(color_)
306 .SetVelocity(spawnVel)
307 .SetRotation(rotation_)
308 .SetRotationVelocity(rotationVelocity_)
309 .SetColorKeys(colorKeys_)
310 .SetSizeKeys(sizeKeys_)
311 .SetUpdateFunction(updateFunc_)
312 .Initialize(particleLifetime_);
313 ++actives_;
314 }

317void Emitter::RegisterGpu() {
318 const auto& ac = Singleton<CameraController>::GetInstance()->GetActive();
319
320 Matrix4x4 rotation;
321 if (billboard_) {
322 rotation = BackToFront * ac->GetMatrix();
323 rotation.matrix[3][0] = 0.f;
324 rotation.matrix[3][1] = 0.f;
325 rotation.matrix[3][2] = 0.f;
326 } else {
327 rotation = MathUtils::Matrix::MakeIdentity();
328 }
329
330 for (uint16_t i = 0; i < actives_; ) {
331 if (particlePool_[i].IsDead()) {
332 --actives_;
333 if (i != actives_) {
334 particlePool_[i] = std::move(particlePool_[actives_]);
335 }
336 continue;
337 }
338 ++i;
339 }
340
341 if (actives_ == 0) return;
342
343 for (uint16_t i = 0; i < actives_; ++i) {
344 Particle& particle = particlePool_[i];
345 particle.Update();
346
347 const Vector3 rot = particle.GetRotation();
348 const Matrix4x4 particleRot = MathUtils::Matrix::MakeRotateX(rot.x)
349 * MathUtils::Matrix::MakeRotateY(rot.y)
350 * MathUtils::Matrix::MakeRotateZ(rot.z);
351
352 mapped_[i].world = MathUtils::Matrix::MakeAffineMatrix(
353 MathUtils::Matrix::MakeScaleMatrix(particle.GetScale()),
354 particleRot * rotation,
355 MathUtils::Matrix::MakeTranslateMatrix(particle.GetPosition())
356 );
357 mapped_[i].wvp = mapped_[i].world * ac->GetViewProjection();
358 mapped_[i].color = particle.GetColor();
359 }
