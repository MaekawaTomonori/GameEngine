194void Emitter::RegisterGpu() {
195 const auto& ac = Singleton<CameraController>::GetInstance()->GetActive();
196 Matrix4x4 billboard = BackToFront * ac->GetMatrix();
197 billboard.matrix[3][0] = 0.f;
198 billboard.matrix[3][1] = 0.f;
199 billboard.matrix[3][2] = 0.f;
200 actives_ = 0;
201
202 std::erase_if(particles_, [&](const auto& _p) { return _p->IsDead(); });
203
204 if (particles_.empty())return;
205
206 for (auto& particle : particles_) {
207 particle->Update();
208 mapped_[actives_].world = MathUtils::Matrix::MakeAffineMatrix(
209 MathUtils::Matrix::MakeScaleMatrix(particle->GetScale()),
210 billboard,
211 MathUtils::Matrix::MakeTranslateMatrix(particle->GetPosition())
212 );
213 mapped_[actives_].wvp = mapped_[actives_].world * ac->GetViewProjection();
214 mapped_[actives_].color = particle->GetColor();
215 ++actives_;
216 }
