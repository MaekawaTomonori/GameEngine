147void ParticleSystem::LoadTemplate(const std::string& _name) {
148 std::string path = PATH + _name + ".json";
149 std::ifstream file(path);
150 if (!file.is_open()) {
151 Log::Send(Log::Level::WARNING, "Failed to open particle template: " + path);
152 return;
153 }
154
155 nlohmann::json root;
156 file >> root;
157
158 Template tmpl;
159 if (root.contains("Emitters") && root["Emitters"].is_array()) {
160 for (const auto& emitterData : root["Emitters"]) {
161 EmitterConfig config;
162 config.texture = emitterData.value("Texture", "white_x16.png");
163 config.frequency = emitterData.value("Frequency", 0.f);
164 config.duration = emitterData.value("Duration", 1.f);
165 config.spawnCount = emitterData.value("SpawnCount", static_cast<uint16_t>(1));
166 config.particleLifetime = emitterData.value("ParticleLifetime", 3.f);
167
168 if (emitterData.contains("Size") && emitterData["Size"].is_array() && emitterData["Size"].size() >= 3) {
169 config.size = {emitterData["Size"][0].get<float>(), emitterData["Size"][1].get<float>(), emitterData["Size"][2].get<float>()};
170 }
171 if (emitterData.contains("Velocity") && emitterData["Velocity"].is_array() && emitterData["Velocity"].size() >= 3) {
172 config.velocity = {emitterData["Velocity"][0].get<float>(), emitterData["Velocity"][1].get<float>(), emitterData["Velocity"][2].get<float>()};
173 }
174 if (emitterData.contains("Color") && emitterData["Color"].is_array() && emitterData["Color"].size() >= 4) {
175 config.color = {emitterData["Color"][0].get<float>(), emitterData["Color"][1].get<float>(), emitterData["Color"][2].get<float>(), emitterData["Color"][3].get<float>()};
176 }
177 if (emitterData.contains("ColorKeys") && emitterData["ColorKeys"].is_array()) {
178 for (const auto& keyData : emitterData["ColorKeys"]) {
179 if (!keyData.contains("Value") || !keyData["Value"].is_array() || keyData["Value"].size() < 4) continue;
180 GradientKey<Vector4> key;
181 key.time = keyData.value("Time", 0.f);
182 key.value = {keyData["Value"][0].get<float>(), keyData["Value"][1].get<float>(), keyData["Value"][2].get<float>(), keyData["Value"][3].get<float>()};
183 config.colorKeys.push_back(key);
184 }
185 }
186 if (emitterData.contains("SizeKeys") && emitterData["SizeKeys"].is_array()) {
187 for (const auto& keyData : emitterData["SizeKeys"]) {
188 if (!keyData.contains("Value") || !keyData["Value"].is_array() || keyData["Value"].size() < 3) continue;
189 GradientKey<Vector3> key;
190 key.time = keyData.value("Time", 0.f);
191 key.value = {keyData["Value"][0].get<float>(), keyData["Value"][1].get<float>(), keyData["Value"][2].get<float>()};
192 config.sizeKeys.push_back(key);
193 }
194 }
195
196 config.updateFuncKey = emitterData.value("UpdateFunc", "");
197 config.spawnFuncKey = emitterData.value("SpawnFunc", "");
198 config.primitive = StringToPrimitive(emitterData.value("Primitive", "Billboard"));
199 config.billboard = emitterData.value("Billboard", true);
200 if (emitterData.contains("Rotation") && emitterData["Rotation"].is_array() && emitterData["Rotation"].size() >= 3) {
201 config.rotation = {emitterData["Rotation"][0].get<float>(), emitterData["Rotation"][1].get<float>(), emitterData["Rotation"][2].get<float>()};
202 }
203 if (emitterData.contains("RotationVelocity") && emitterData["RotationVelocity"].is_array() && emitterData["RotationVelocity"].size() >= 3) {
204 config.rotationVelocity = {emitterData["RotationVelocity"][0].get<float>(), emitterData["RotationVelocity"][1].get<float>(), emitterData["RotationVelocity"][2].get<float>()};
205 }
206 tmpl.emitters.push_back(config);
207 }
208 }
209
210 templates_[_name] = std::move(tmpl);
211 Log::Send(Log::Level::INFO, "Loaded particle template: " + _name);

216 Log::Send(Log::Level::WARNING, "Template not found for saving: " + _name);

220 const auto& tmpl = templates_.at(_name);

242 emitterData["ColorKeys"] = colorKeysJson;
243 }

249 {"Value", {key.value.x, key.value.y, key.value.z}}
250 });
251 }
252 emitterData["SizeKeys"] = sizeKeysJson;
253 }
254 if (!config.updateFuncKey.empty()) {
255 emitterData["UpdateFunc"] = config.updateFuncKey;

257 if (!config.spawnFuncKey.empty()) {
258 emitterData["SpawnFunc"] = config.spawnFuncKey;

261 emitterData["Billboard"] = config.billboard;

264 emittersJson.push_back(emitterData);
265 }

267 root["Emitters"] = emittersJson;
268
269 std::filesystem::create_directories("Assets/Data/Particle/");
270 std::string path = PATH + _name + ".json";
271 std::ofstream file(path);
272 if (!file.is_open()) {
273 Log::Send(Log::Level::WARNING, "Failed to save particle template: " + path);
274 return;
275 }

281EmitterHandle ParticleSystem::Emit(const std::string& _templateName, const Vector3& _position) {
282 if (!poolInitialized_) {
283 InitializePool();
284 }
285
286 if (!templates_.contains(_templateName)) {
287 Log::Send(Log::Level::WARNING, "Template not found: " + _templateName);
288 return EmitterHandle();

292 EmitterHandle firstHandle;
293
294 for (const auto& config : tmpl.emitters) {
295 if (available_.empty()) {

297 return firstHandle;
298 }
299
300 Emitter* emitter = available_.back();
301 available_.pop_back();
302

304 .SetFrequency(config.frequency)
305 .SetDuration(config.duration)

307 .SetSize(config.size)

309 .SetColor(config.color)
310 .SetPosition(_position)
311 .SetBillboard(config.billboard)
312 .SetPrimitive(config.primitive)
313 .SetRotation(config.rotation)
314 .SetRotationVelocity(config.rotationVelocity)
315 .SetParticleLifetime(config.particleLifetime)
316 .SetColorKeys(config.colorKeys)
317 .SetSizeKeys(config.sizeKeys);
318

320 if (updateFunc) {

540 rangeForVS[0].BaseShaderRegister = 0;
541 rangeForVS[0].NumDescriptors = 1;
542 rangeForVS[0].RangeType = D3D12_DESCRIPTOR_RANGE_TYPE_SRV;
543 rangeForVS[0].OffsetInDescriptorsFromTableStart = D3D12_DESCRIPTOR_RANGE_OFFSET_APPEND;

545 D3D12_DESCRIPTOR_RANGE rangeForPS[1]{};
546 rangeForPS[0].BaseShaderRegister = 0;

549 rangeForPS[0].OffsetInDescriptorsFromTableStart = D3D12_DESCRIPTOR_RANGE_OFFSET_APPEND;

551 RootSignature rs = RootSignature();
552 rs.AddParameter({
553 .ParameterType = D3D12_ROOT_PARAMETER_TYPE_DESCRIPTOR_TABLE,
554 .DescriptorTable = {
555 .NumDescriptorRanges = _countof(rangeForVS),
556 .pDescriptorRanges = rangeForVS,

568 .AddressU = D3D12_TEXTURE_ADDRESS_MODE_WRAP,
569 .AddressV = D3D12_TEXTURE_ADDRESS_MODE_WRAP,
570 .AddressW = D3D12_TEXTURE_ADDRESS_MODE_WRAP,
571 .ComparisonFunc = D3D12_COMPARISON_FUNC_NEVER,
572 .MaxLOD = D3D12_FLOAT32_MAX,
573 .ShaderRegister = 0,
574 .ShaderVisibility = D3D12_SHADER_VISIBILITY_PIXEL,

578 il.SetElement({

585 .SemanticName = "TEXCOORD",
586 .SemanticIndex = 0,
