113 float texLeft = leftTop_.x / static_cast<float>(metadata.width);
114 float texRight = (leftTop_.x + texSize_.x) / static_cast<float>(metadata.width);
115 float texTop = leftTop_.y / static_cast<float>(metadata.height);
116 float texBottom = (leftTop_.y + texSize_.y) / static_cast<float>(metadata.height);
117
118 vd_[0].uv= {texLeft, texBottom};
119 vd_[1].uv= {texLeft, texTop};
120 vd_[2].uv= {texRight, texBottom};
121 vd_[3].uv= {texRight, texTop};
122#pragma endregion
123
124 wd_->world = MathUtils::Matrix::MakeAffineMatrix({size_.x, size_.y, 1}, Vector3{0, 0, rotation_}, {position_.x, position_.y, 0});

127 wd_->wvp = (wd_->world * viewProjection);
