366IPostEffect* PostProcessExecutor::FindOrCreate(const std::string& _type, const std::string& _name, bool _create) {
367 // 既存 instance を検索する。
368 for (auto& effectData : effects_) {
369 if (effectData.name == _name) {
370 return effectData.effect.get();
371 }
372 }
373
374 // 自動生成しない場合はここで終了。
375 if (!_create) return nullptr;
376
377 // Factory が未設定なら生成できない。
378 if (!factory_) {
379 Log::Send(Log::Level::ERR, "PostEffectFactory is not set");
380 return nullptr;
381 }
382
383 // Factory から新しい effect を生成する。
384 auto newEffect = factory_->Create(_type);
385 if (!newEffect) {
386 Log::Send(Log::Level::ERR, std::format("Failed to create effect type: {}", _type));
387 return nullptr;
388 }
389
390 // effect を executor 配下へ組み込む。
391 newEffect->SetUp(adapter_, srv_);
392 auto rtvHandle = rtvHeap_->GetCPUHandle(static_cast<uint32_t>(effects_.size()) + 1);
393 newEffect->SetRTVHandle(rtvHandle);
394 newEffect->Initialize();
395
396 effects_.emplace_back(EffectData{ std::move(newEffect), _name, _type, true });
397
398 return effects_.back().effect.get();

401void PostProcessExecutor::ApplyPreset(const std::string& _presetName, const std::string& _mode, const std::vector<std::string>& _ignoreList, std::function<void()> _onComplete) {
402 // 完了時コールバックを保持する。
403 onAnimationComplete_ = _onComplete;
404
405 std::string path = "./Assets/Data/PostEffect/presets.json";
406
407 // preset 定義ファイルの存在確認。
408 if (!std::filesystem::exists(path)) {
409 Log::Send(Log::Level::ERR, std::format("Preset file not found: {}", path));
410 return;
411 }
412
413 std::ifstream file(path);
414 if (!file.is_open()) {
415 Log::Send(Log::Level::ERR, std::format("Failed to open preset file: {}", path));
416 return;
417 }
418
419 json presetsJson;
420 file >> presetsJson;
421 file.close();
422
423 // 指定 preset が存在するか確認する。
424 if (!presetsJson.contains(_presetName)) {
425 Log::Send(Log::Level::WARNING, std::format("Preset '{}' not found in presets.json", _presetName));
426 return;
427 }
428
429 const auto& presetData = presetsJson[_presetName];
430
431 // アニメーション時間を読み取る。
432 animationDuration_ = presetData.value("duration", 0.0f);
433
434 // replace モードでは既存 effect を一旦無効化する。
435 if (_mode == "replace") {
436 for (auto& effectData : effects_) {
437 effectData.enabled = false;
438 }
439 }
440
441 // 今回のアニメーション対象を初期化する。
442 animatingEffects_.clear();
443
444 // effects / members のどちらにも対応する。
445 const char* arrayKey = presetData.contains("effects") ? "effects" : "members";
446
447 if (presetData.contains(arrayKey)) {
448 for (const auto& effectEntry : presetData[arrayKey]) {
449 std::string type = effectEntry["type"];
450 std::string name = effectEntry["name"];
451
452 // preset 名省略時は現在の preset 名を使う。
453 std::string presetName = effectEntry.value("preset", _presetName);
454 bool autoCreate = effectEntry.value("autoCreate", true);
455
456 // ignore list に含まれる effect はスキップする。
457 bool shouldIgnore = false;
458 for (const auto& ignoreName : _ignoreList) {
459 if (ignoreName == name) {
460 shouldIgnore = true;
461 break;
462 }
463 }
464 if (shouldIgnore) continue;
465
466 // effect を取得または生成する。
467 IPostEffect* effect = FindOrCreate(type, name, autoCreate);
468 if (!effect) {
469 Log::Send(Log::Level::WARNING, std::format("Failed to create effect: {} ({})", name, type));
470 continue;
471 }
472
473 // preset を読み込み、有効化する。
474 effect->LoadPreset(presetName);
475 Log::Send(Log::Level::DBG, std::format("Loaded preset '{}' for effect '{}' ({})", presetName, name, type));
476
477 SetActive(name, true);
478 Log::Send(Log::Level::DBG, std::format("Enabled effect '{}' (enabled: {})", name, true));
479
480 // アニメーション対象として記録する。
481 animatingEffects_.push_back(name);
482 }
483 }
484
485 // アニメーションありの場合は更新ループへ移行する。
486 if (animationDuration_ > 0.0f) {
487 isAnimating_ = true;
488 animationTimer_ = 0.0f;
489 Log::Send(Log::Level::INFO, std::format("Started animation for preset '{}' (duration: {:.1f}s, effects: {})",
490 _presetName, animationDuration_, animatingEffects_.size()));
491 } else {
492 // duration が 0 の場合は即座に最終状態を適用する。
493 for (const auto& effectName : animatingEffects_) {
494 for (auto& effectData : effects_) {
495 if (effectData.name == effectName) {
496 effectData.effect->UpdateAnimation(1.0f);
497 break;
498 }
499 }
500 }
501 isAnimating_ = false;
502 }
503
504 // デバッグ出力。
505 Log::Send(Log::Level::INFO, std::format("Total effects: {}", effects_.size()));
506 for (const auto& effectData : effects_) {
507 Log::Send(Log::Level::INFO, std::format(" Effect '{}' ({}): enabled={}",
508 effectData.name, effectData.type, effectData.enabled ? "true" : "false"));
509 }
510
511 Log::Send(Log::Level::INFO, std::format("Preset '{}' applied with mode '{}'", _presetName, _mode));
512}

514void PostProcessExecutor::Update(float _deltaTime) {
515 if (!isAnimating_) return;
516
517 animationTimer_ += _deltaTime;
518 float t = std::min(animationTimer_ / animationDuration_, 1.0f);
519
520 // アニメーション対象 effect を進行度 t で更新する。
521 for (const auto& effectName : animatingEffects_) {
522 for (auto& effectData : effects_) {
523 if (effectData.name == effectName) {
524 effectData.effect->UpdateAnimation(t);
525 break;
526 }
527 }
528 }
529
530 // 最後まで到達したら状態を片付ける。
531 if (t >= 1.0f) {
532 isAnimating_ = false;
533
534 // 一時的に使った effect を初期状態へ戻して無効化する。
535 for (const auto& effectName : animatingEffects_) {
536 for (auto& effectData : effects_) {
537 if (effectData.name == effectName) {
538 effectData.enabled = false;
539 effectData.effect->Initialize();
540
541 Log::Send(Log::Level::INFO, std::format("Reset and disabled effect '{}'", effectName));
542 break;
543 }
544 }
545 }
546
547 // 対象一覧をクリアする。
548 animatingEffects_.clear();
549
550 Log::Send(Log::Level::INFO, "PostEffect animation completed and effects reset");
551
552 // 完了コールバックがあれば呼び出す。
553 if (onAnimationComplete_) {
554 Log::Send(Log::Level::INFO, "Calling animation complete callback");
555 onAnimationComplete_();
556 onAnimationComplete_ = nullptr;
557 }
558 }

582

592 inFile >> allPresets;
593 inFile.close();
594 }
595 }
596
597 // 新しい preset を上書きまたは追加する。
598 allPresets[_presetName] = presetJson;

600 // ファイルへ書き戻す。
