306void PostProcessExecutor::ApplyPreset(const std::string& _presetName, std::function<void()> _onComplete) {
307 // 完了時コールバックを保持する。
308 onAnimationComplete_ = _onComplete;
309
310 std::string path = PresetsFilePath();
311
312 // preset 定義ファイルの存在確認。
313 if (!std::filesystem::exists(path)) {
314 Log::Send(Log::Level::ERR, std::format("Preset file not found: {}", path));
315 return;
316 }
317
318 std::ifstream file(path);
319 if (!file.is_open()) {
320 Log::Send(Log::Level::ERR, std::format("Failed to open preset file: {}", path));
321 return;
322 }
323
324 json presetsJson;
325 file >> presetsJson;
326 file.close();
327
328 // 指定 preset が存在するか確認する。
329 if (!presetsJson.contains(_presetName)) {
330 Log::Send(Log::Level::WARNING, std::format("Preset '{}' not found in presets.json", _presetName));
331 return;
332 }
333
334 const auto& presetData = presetsJson[_presetName];
335
336 // アニメーション時間を読み取る。
337 animationDuration_ = presetData.value("duration", 0.0f);
338
339 // 今回のアニメーション対象（一時エフェクトのid）を初期化する。
340 animatingTemporaries_.clear();
341
342 if (presetData.contains("members")) {
343 for (const auto& memberEntry : presetData["members"]) {
344 std::string type = memberEntry["type"];
345
346 // preset 名省略時は現在の preset 名を使う。
347 std::string presetName = memberEntry.value("preset", _presetName);
348
349 // 演出用の一時エフェクトを生成する。
350 const uint32_t id = CreateTemporary(type);
351 if (id == 0) {
352 Log::Send(Log::Level::WARNING, std::format("Failed to create temporary effect: {}", type));
353 continue;
354 }
355
356 IPostEffect* effect = GetTemporary(id);
357 effect->LoadPreset(presetName);
358 Log::Send(Log::Level::DBG, std::format("Loaded preset '{}' for temporary effect ({})", presetName, type));
359
360 animatingTemporaries_.push_back(id);
361 }
362 }
363
364 // アニメーションありの場合は更新ループへ移行する。
365 if (animationDuration_ > 0.0f) {
366 isAnimating_ = true;
367 animationTimer_ = 0.0f;
368 Log::Send(Log::Level::INFO, std::format("Started animation for preset '{}' (duration: {:.1f}s, effects: {})",
369 _presetName, animationDuration_, animatingTemporaries_.size()));
370 } else {
371 // duration が 0 の場合は即座に最終状態を適用してから破棄する。
372 for (const uint32_t id : animatingTemporaries_) {
373 if (IPostEffect* effect = GetTemporary(id)) {
374 effect->UpdateAnimation(1.0f);
375 }
376 RemoveTemporary(id);
377 }
378 animatingTemporaries_.clear();
379 isAnimating_ = false;
380
381 if (onAnimationComplete_) {
382 onAnimationComplete_();
383 onAnimationComplete_ = nullptr;
384 }
385 }
386
387 Log::Send(Log::Level::INFO, std::format("Preset '{}' applied", _presetName));
388}

390void PostProcessExecutor::Update(float _deltaTime) {
391 if (!isAnimating_) return;
392
393 animationTimer_ += _deltaTime;
394 float t = std::min(animationTimer_ / animationDuration_, 1.0f);
395
396 // アニメーション対象の一時エフェクトを進行度 t で更新する。
397 for (const uint32_t id : animatingTemporaries_) {
398 if (IPostEffect* effect = GetTemporary(id)) {
399 effect->UpdateAnimation(t);
400 }
401 }
402
403 // 最後まで到達したら一時エフェクトを破棄する。
404 if (t >= 1.0f) {
405 isAnimating_ = false;
406
407 for (const uint32_t id : animatingTemporaries_) {
408 RemoveTemporary(id);
409 }
410 animatingTemporaries_.clear();
411
412 Log::Send(Log::Level::INFO, "PostEffect animation completed and temporary effects removed");
413
414 // 完了コールバックがあれば呼び出す。
415 if (onAnimationComplete_) {
416 Log::Send(Log::Level::INFO, "Calling animation complete callback");
417 onAnimationComplete_();
418 onAnimationComplete_ = nullptr;
419 }
420 }

423void PostProcessExecutor::SavePreset(const std::string& _presetName) {
424 json presetJson;
425
426 // 現在有効な effect だけを書き出す。
427 for (const auto& effectData : chain_->GetEffects()) {
428 if (effectData.enabled) {
429 json memberEntry;
430 memberEntry["type"] = effectData.type;
431 memberEntry["preset"] = _presetName;
432
433 presetJson["members"].push_back(memberEntry);
434
435 // 各 effect のパラメータも保存する。
436 effectData.effect->SavePreset(_presetName);
437 }
438 }
439
440 // duration を保存する。
441 presetJson["duration"] = animationDuration_;
442
443 // presets.json を更新する。
444 std::string presetsPath = PresetsFilePath();
445 std::filesystem::create_directories("./Assets/Data/PostEffect/" + name_);
446
447 // 既存ファイルがあれば読み込む。
448 json allPresets;
449 if (std::filesystem::exists(presetsPath)) {
450 std::ifstream inFile(presetsPath);
451 if (inFile.is_open()) {
452 inFile >> allPresets;
453 inFile.close();
454 }
455 }
456
457 // 新しい preset を上書きまたは追加する。
458 allPresets[_presetName] = presetJson;
459
460 // ファイルへ書き戻す。
461 std::ofstream outFile(presetsPath);
462 if (!outFile.is_open()) {
463 Log::Send(Log::Level::ERR, std::format("Failed to save preset to: {}", presetsPath));
464 return;
465 }
466
467 outFile << allPresets.dump(4);
468 outFile.close();
469
470 Log::Send(Log::Level::INFO, std::format("Preset '{}' saved successfully", _presetName));
471}
