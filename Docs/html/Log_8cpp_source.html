28void Log::Initialize() {
29 // Initialize execution context information
30 InitializeExecutionContext();
31
32 // Clear any existing sinks
33 sinks_.clear();
34
35 // Console sink with execution context in pattern
36 auto console = std::make_shared<spdlog::sinks::stdout_color_sink_mt>();
37 console->set_pattern("%^[%T] [%l] [Engine] [" + workingDirectory_ + "] %v%$");
38 sinks_.emplace_back(console);
39
40 // Daily rotating file sink with full execution context
41 auto file = std::make_shared<spdlog::sinks::daily_file_sink_mt>(logFilePath_ + logFileName_ + logFileExt_, 0, 0);
42 file->set_pattern("[%Y-%m-%d %T] [%l] [Engine] [EXE:" + executablePath_ + "] [CWD:" + workingDirectory_ + "] %v");
43 sinks_.emplace_back(file);
44
45#ifdef _WIN32
46 // Visual Studio output window sink with execution context
47 auto windows_sink = std::make_shared<spdlog::sinks::msvc_sink_mt>();
48 windows_sink->set_pattern("[%T] [%l] [Engine] [" + workingDirectory_ + "] %v");
49 sinks_.emplace_back(windows_sink);
50#endif
51
52 // Create logger with all sinks
53 auto logger = std::make_shared<spdlog::logger>("Engine", begin(sinks_), end(sinks_));
54 logger->set_level(static_cast<spdlog::level::level_enum>(static_cast<int>(level_)));
55
56 // Register as default logger
57 spdlog::set_default_logger(logger);
58
59 // Set up async logging
60 spdlog::flush_every(std::chrono::seconds(1));
61
62 spdlog::info("Log system initialized");
63
64 // Log execution context immediately after initialization
65 LogExecutionContext();
66}

139void Log::LogFileOperation(const std::string& operation, const std::string& filePath, bool success, const std::string& details) {
140 std::string status = success ? "SUCCESS" : "FAILED";
141 std::string message = std::format("FILE_OP [{}] {} -> {}", status, operation, filePath);
142
143 if (!details.empty()) {
144 message += " (" + details + ")";
145 }

147 // Check if file exists and add absolute path info

151 message += std::format(" [ABS: {}]", absolutePath);

153 message += " [FILE_NOT_FOUND]";

155 } catch (const std::exception& e) {
156 message += std::format(" [PATH_ERROR: {}]", e.what());
157 }
158
159 Level logLevel = success ? Level::INFO : Level::ERR;
160 Send(logLevel, message);
