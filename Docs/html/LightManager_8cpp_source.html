96void LightManager::Load() {
97 Json* json = Singleton<Json>::GetInstance();
98 json->Load(path);
99
100 rawDirectionalLights_.clear();
101 rawPointLights_.clear();
102 rawSpotLights_.clear();
103 *lightCount_ = {0,0,0};
104 auto data = json->GetGroups(path);
105 for (auto& itr : data) {
106 auto group = itr.second;
107 switch (magic_enum::enum_value<LightType>(std::get<int32_t>(group.at("type")))){
108 case LightType::Directional:
109 Add(LightType::Directional);
110 rawDirectionalLights_.back()->Set(itr.first, {
111 std::get<Vector4>(group.at("color")),
112 std::get<Vector3>(group.at("direction")),
113 std::get<float>(group.at("intensity"))
114 });
115 break;
116 case LightType::Point:
117 Add(LightType::Point);
118 rawPointLights_.back()->Set(itr.first, {
119 std::get<Vector4>(group.at("color")),
120 std::get<Vector3>(group.at("position")),
121 std::get<float>(group.at("intensity")),
122 std::get<float>(group.at("radius")),
123 std::get<float>(group.at("decay")),
124 {0,0}
125 });
126 break;
127 case LightType::Spot:
128 Add(LightType::Spot);
129 rawSpotLights_.back()->Set(itr.first, {
130 std::get<Vector4>(group.at("color")),
131 std::get<Vector3>(group.at("position")),
132 std::get<float>(group.at("intensity")),
133 std::get<Vector3>(group.at("direction")).Normalize(),
134 std::get<float>(group.at("distance")),
135 std::get<float>(group.at("decay")),
136 std::get<float>(group.at("cosAngle")),
137 std::get<float>(group.at("falloffStart")),
138 0
139 });
140 break;
141 }
142 }
