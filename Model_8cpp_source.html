313void Model::SetBindPose(Skeleton& _skeleton) {
314 size_t jointSize = _skeleton.joints.size();
315 skinCluster_.inverseBindPoses.resize(jointSize);
316 for (auto& inverseBindPose : skinCluster_.inverseBindPoses){
317 inverseBindPose = MathUtils::Matrix::MakeIdentity();
318 }
319
320 for (const auto& jointWeight : data_->skinCluster){
321 auto itr = _skeleton.map.find(jointWeight.first);
322 if (itr == _skeleton.map.end()){
323 continue;
324 }
325 skinCluster_.inverseBindPoses[itr->second] = jointWeight.second.inverseBindPose;
326 for (const auto& vertexWeight : jointWeight.second.weights){
327 auto& currentInfluence = skinCluster_.mappedInfluence[vertexWeight.index];
328 for (uint32_t index = 0; index < MAX_INFLUENCE; ++index){
329 if (currentInfluence.weights[index] == 0.f){
330 currentInfluence.weights[index] = vertexWeight.weight;
331 currentInfluence.jointIndices[index] = static_cast<int32_t>(itr->second);
332 break;
333 }
334 }
335 }
336 }

339void Model::UpdateSkinCluster() {
340 if (!data_->skeleton.has_value() || data_->skinCluster.empty()) {
341 Log::Send(Log::Level::WARNING, "Model data does not contain valid skinning data, skipping skin cluster update");
342 return;
343 }
344
345 Skeleton& skeleton = data_->skeleton.value();
346 for (size_t jointIndex = 0; jointIndex < skeleton.joints.size(); ++jointIndex) {
347 if (skinCluster_.inverseBindPoses.size() <= jointIndex) {
348 Utils::Alert("Joint index out of bounds in skin cluster update");
349 break;
350 }
351
352 skinCluster_.mappedPalette[jointIndex].space = skinCluster_.inverseBindPoses[jointIndex] * skeleton.joints[jointIndex].space;
353 skinCluster_.mappedPalette[jointIndex].inverseTranspose = skinCluster_.mappedPalette[jointIndex].space.Inverse().Transpose();
354 }
