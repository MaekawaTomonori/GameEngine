215void Model::Load(const std::string& _name) {
216 auto repo = Singleton<ModelCommon>::GetInstance()->GetResourceRepository();
217 if (repo->GetModelRepository()->Contains(_name)){
218 Log::Send(Log::Level::WARNING, "Already Loaded Model : " + _name);
219 return;
220 }
221
222 std::unique_ptr<IModelLoader> loader;
223 if (std::filesystem::exists("Assets/Resources/" + _name + "/" + _name + ".obj")) {
224 loader = std::make_unique<ObjLoader>();
225 }
226 else if (std::filesystem::exists("Assets/Resources/" + _name + "/" + _name + ".gltf")) {
227 loader = std::make_unique<GltfLoader>();
228 }
229 else {
230 Log::Send(Log::Level::ERR, "Model::Load: Model not found: " + _name);
231 Utils::Alert("Model not found: " + _name);
232 return;
233 }
234 loader->LoadModel(_name, repo);
235}
