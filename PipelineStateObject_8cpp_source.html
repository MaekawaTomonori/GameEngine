26PipelineStateObject& PipelineStateObject::SetBlend(const BlendMode _blendMode) {
27 blendDesc_.RenderTarget[0].BlendEnable = true;
28 blendDesc_.RenderTarget[0].RenderTargetWriteMask = D3D12_COLOR_WRITE_ENABLE_ALL;
29 switch (_blendMode){
30 case BlendMode::ALPHA:
31 blendDesc_.RenderTarget[0].SrcBlend = D3D12_BLEND_SRC_ALPHA;
32 blendDesc_.RenderTarget[0].BlendOp = D3D12_BLEND_OP_ADD;
33 blendDesc_.RenderTarget[0].DestBlend = D3D12_BLEND_INV_SRC_ALPHA;
34 // アルファチャンネルはデスティネーション値（1.0）を維持する
35 // result.a = src.a * 0 + dst.a * 1 = dst.a
36 // → renderTexture_ のアルファが透過スプライトで 0 になることを防ぐ
37 blendDesc_.RenderTarget[0].SrcBlendAlpha = D3D12_BLEND_ZERO;
38 blendDesc_.RenderTarget[0].BlendOpAlpha = D3D12_BLEND_OP_ADD;
39 blendDesc_.RenderTarget[0].DestBlendAlpha = D3D12_BLEND_ONE;
40 break;
41 case BlendMode::ADD:
42 blendDesc_.RenderTarget[0].SrcBlend = D3D12_BLEND_SRC_ALPHA;
43 blendDesc_.RenderTarget[0].BlendOp = D3D12_BLEND_OP_ADD;
44 blendDesc_.RenderTarget[0].DestBlend = D3D12_BLEND_ONE;
45 blendDesc_.RenderTarget[0].SrcBlendAlpha = D3D12_BLEND_ZERO;
46 blendDesc_.RenderTarget[0].DestBlendAlpha = D3D12_BLEND_ONE;
47 blendDesc_.RenderTarget[0].BlendOpAlpha = D3D12_BLEND_OP_ADD;
48 break;
49 case BlendMode::SUB:
50 blendDesc_.RenderTarget[0].SrcBlend = D3D12_BLEND_SRC_ALPHA;
51 blendDesc_.RenderTarget[0].BlendOp = D3D12_BLEND_OP_REV_SUBTRACT;
52 blendDesc_.RenderTarget[0].DestBlend = D3D12_BLEND_ONE;
53 blendDesc_.RenderTarget[0].SrcBlendAlpha = D3D12_BLEND_ZERO;
54 blendDesc_.RenderTarget[0].BlendOpAlpha = D3D12_BLEND_OP_ADD;
55 blendDesc_.RenderTarget[0].DestBlendAlpha = D3D12_BLEND_ONE;
56 break;
57 case BlendMode::MULTI:
58 blendDesc_.RenderTarget[0].SrcBlend = D3D12_BLEND_ZERO;
59 blendDesc_.RenderTarget[0].BlendOp = D3D12_BLEND_OP_ADD;
60 blendDesc_.RenderTarget[0].DestBlend = D3D12_BLEND_DEST_COLOR;
61 blendDesc_.RenderTarget[0].SrcBlendAlpha = D3D12_BLEND_ZERO;
62 blendDesc_.RenderTarget[0].BlendOpAlpha = D3D12_BLEND_OP_ADD;
63 blendDesc_.RenderTarget[0].DestBlendAlpha = D3D12_BLEND_ONE;
64 break;
65 case BlendMode::SCREEN:
66 blendDesc_.RenderTarget[0].SrcBlend = D3D12_BLEND_INV_DEST_COLOR;
67 blendDesc_.RenderTarget[0].BlendOp = D3D12_BLEND_OP_ADD;
68 blendDesc_.RenderTarget[0].DestBlend = D3D12_BLEND_ONE;
69 blendDesc_.RenderTarget[0].SrcBlendAlpha = D3D12_BLEND_ZERO;
70 blendDesc_.RenderTarget[0].BlendOpAlpha = D3D12_BLEND_OP_ADD;
71 blendDesc_.RenderTarget[0].DestBlendAlpha = D3D12_BLEND_ONE;
72 break;
73 case BlendMode::NONE:
74 blendDesc_.RenderTarget[0].BlendEnable = false;
75 blendDesc_.RenderTarget[0].RenderTargetWriteMask = D3D12_COLOR_WRITE_ENABLE_ALL;
76 }
77
78 return *this;
79}

112void PipelineStateObject::Create() {
113 desc_.pRootSignature = rootSignature_.Get();
114 desc_.InputLayout = inputLayout_.Get();
115 desc_.BlendState = blendDesc_;
116 desc_.VS = { shader_->GetVertexShader()->GetBufferPointer(), shader_->GetVertexShader()->GetBufferSize() };
117 desc_.RasterizerState = rasterizer_;
118 desc_.PS = { shader_->GetPixelShader()->GetBufferPointer(), shader_->GetPixelShader()->GetBufferSize() };
119 desc_.DepthStencilState = depthStencilDesc_;
120
121 desc_.NumRenderTargets = numRenderTargets_;
122 desc_.RTVFormats[0] = (numRenderTargets_ > 0) ? rtvFormat_ : DXGI_FORMAT_UNKNOWN;
123
124 desc_.PrimitiveTopologyType = topologyType_;
125 desc_.SampleDesc.Count = 1;
126 desc_.SampleMask = D3D12_DEFAULT_SAMPLE_MASK;
127 // Setting more options
128
129 HRESULT hr = adapter_->GetDevice()->CreateGraphicsPipelineState(&desc_, IID_PPV_ARGS(&pso_));
130 if (FAILED(hr)){
131 Log::Send(Log::Level::ERR, "Failed to create pipeline state object");
132 Utils::Alert("Failed to create pipeline state object");
133 return;
134 }
135 Log::Send(Log::Level::INFO, "Pipeline state object created successfully");
136}
