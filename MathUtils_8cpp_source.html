44Vector3 MathUtils::Matrix::Transform(const Vector3& _vector, const Matrix4x4& _matrix) {

46 _vector.x * _matrix.matrix[0][0] + _vector.y * _matrix.matrix[1][0] + _vector.z * _matrix.matrix[2][0] + 1 * _matrix.matrix[3][0],
47 _vector.x * _matrix.matrix[0][1] + _vector.y * _matrix.matrix[1][1] + _vector.z * _matrix.matrix[2][1] + 1 * _matrix.matrix[3][1],

97Matrix4x4 MathUtils::Matrix::MakeRotate(const Quaternion& _rotate) {
98 float x2 = _rotate.x * _rotate.x;
99 float y2 = _rotate.y * _rotate.y;
100 float z2 = _rotate.z * _rotate.z;
101 float xy = _rotate.x * _rotate.y;
102 float xz = _rotate.x * _rotate.z;
103 float xw = _rotate.x * _rotate.w;
104 float yz = _rotate.y * _rotate.z;
105 float yw = _rotate.y * _rotate.w;
106 float zw = _rotate.z * _rotate.w;
107 return {
108 1.f - 2.f * (y2 + z2), 2.f * (xy + zw), 2.f * (xz - yw), 0.f,
109 2.f * (xy - zw), 1.f - 2.f * (x2 + z2), 2.f * (yz + xw), 0.f,
110 2.f * (xz + yw), 2.f * (yz - xw), 1.f - 2.f * (x2 + y2), 0.f,
111 0.f, 0.f, 0.f, 1.f
112 };
113}

119Matrix4x4 MathUtils::Matrix::MakeAffineMatrix(const Vector3& _scale, const Vector3& _rotate, const Vector3& _translate) {

209Quaternion MathUtils::Slerp(const Quaternion& _a, const Quaternion& _b, const float _t) {
210 float dot = _a.x * _b.x + _a.y * _b.y + _a.z * _b.z + _a.w * _b.w;
211 float epsilon = 1e-6f;
212
213 Quaternion b_temp = _b;
214 if (dot < 0){
215 dot = -dot;
216 b_temp = { -_b.x, -_b.y, -_b.z, -_b.w };
217 }
218 if (1 - dot < epsilon){
219 return (_a * (1.0f - _t) + b_temp * _t).Normalize();
220 }
221 float theta = acosf(dot);
222 float sinTheta = sinf(theta);
223 float s0 = sinf((1 - _t) * theta) / sinTheta;
224 float s1 = sinf(_t * theta) / sinTheta;
225 return (_a * s0 + b_temp * s1).Normalize();
226}
