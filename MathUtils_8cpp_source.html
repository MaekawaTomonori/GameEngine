95Matrix4x4 MathUtils::Matrix::MakeRotate(const Quaternion& rotate) {
96 float x2 = rotate.x * rotate.x;
97 float y2 = rotate.y * rotate.y;
98 float z2 = rotate.z * rotate.z;
99 float xy = rotate.x * rotate.y;
100 float xz = rotate.x * rotate.z;
101 float xw = rotate.x * rotate.w;
102 float yz = rotate.y * rotate.z;
103 float yw = rotate.y * rotate.w;
104 float zw = rotate.z * rotate.w;
105 return {
106 1.f - 2.f * (y2 + z2), 2.f * (xy + zw), 2.f * (xz - yw), 0.f,
107 2.f * (xy - zw), 1.f - 2.f * (x2 + z2), 2.f * (yz + xw), 0.f,
108 2.f * (xz + yw), 2.f * (yz - xw), 1.f - 2.f * (x2 + y2), 0.f,
109 0.f, 0.f, 0.f, 1.f
110 };
111}

204Quaternion MathUtils::Slerp(const Quaternion& a, const Quaternion& b, const float t) {
205 float dot = a.x * b.x + a.y * b.y + a.z * b.z + a.w * b.w;
206 float epsilon = 1e-6f;
207
208 Quaternion b_temp = b;
209 if (dot < 0){
210 dot = -dot;
211 b_temp = { -b.x, -b.y, -b.z, -b.w };
212 }
213 if (1 - dot < epsilon){
214 return (a * (1.0f - t) + b_temp * t).Normalize();
215 }
216 float theta = acosf(dot);
217 float sinTheta = sinf(theta);
218 float s0 = sinf((1 - t) * theta) / sinTheta;
219 float s1 = sinf(t * theta) / sinTheta;
220 return (a * s0 + b_temp * s1).Normalize();
221}
