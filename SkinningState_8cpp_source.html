131void SkinningState::SetBindPose() {
132 size_t jointSize = pose_.joints.size();
133 skinCluster_.inverseBindPoses.resize(jointSize);
134 for (auto& inverseBindPose : skinCluster_.inverseBindPoses) {
135 inverseBindPose = MathUtils::Matrix::MakeIdentity();
136 }
137
138 for (const auto& jointWeight : data_->skinCluster) {
139 auto itr = pose_.map.find(jointWeight.first);
140 if (itr == pose_.map.end()) {
141 continue;
142 }
143 skinCluster_.inverseBindPoses[itr->second] = jointWeight.second.inverseBindPose;
144 for (const auto& vertexWeight : jointWeight.second.weights) {
145 auto& currentInfluence = skinCluster_.mappedInfluence[vertexWeight.index];
146 for (uint32_t index = 0; index < MAX_INFLUENCE; ++index) {
147 if (currentInfluence.weights[index] == 0.f) {
148 currentInfluence.weights[index] = vertexWeight.weight;
149 currentInfluence.jointIndices[index] = static_cast<int32_t>(itr->second);
150 break;
151 }
152 }
153 }

157void SkinningState::UpdateSkinCluster() {
158 for (size_t jointIndex = 0; jointIndex < pose_.joints.size(); ++jointIndex) {
159 if (skinCluster_.inverseBindPoses.size() <= jointIndex) {
