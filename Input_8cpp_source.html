162void Input::ApplyCursorVisibility() {
163 bool shouldShow = cursorVisible_;
164
165 // タイトルバーなどのノンクライアント領域ではカーソルを常に表示する
166 if (!shouldShow) {
167 POINT ptScreen;
168 GetCursorPos(&ptScreen);
169 RECT clientRect;
170 GetClientRect(hWnd_, &clientRect);
171 POINT topLeft = { 0, 0 };
172 ClientToScreen(hWnd_, &topLeft);
173 const bool inClientArea =
174 ptScreen.x >= topLeft.x &&
175 ptScreen.x < topLeft.x + clientRect.right &&
176 ptScreen.y >= topLeft.y &&
177 ptScreen.y < topLeft.y + clientRect.bottom;
178 if (!inClientArea) {
179 shouldShow = true;
180 }
181 }
182
183#ifdef _DEBUG
184 if (!cursorVisible_) {
185 if (sceneViewTransform_.active
186 && sceneViewTransform_.imageSize.x > 0.f
187 && sceneViewTransform_.imageSize.y > 0.f) {
188 // スクリーン座標で SceneView ヒットテスト（imagePos は GetItemRectMin() 由来）
189 POINT pt;
190 GetCursorPos(&pt);
191 ScreenToClient(hWnd_, &pt);
192 const float px = static_cast<float>(pt.x);
193 const float py = static_cast<float>(pt.y);
194 const bool insideScene =
195 px >= sceneViewTransform_.imagePos.x &&
196 px < sceneViewTransform_.imagePos.x + sceneViewTransform_.imageSize.x &&
197 py >= sceneViewTransform_.imagePos.y &&
198 py < sceneViewTransform_.imagePos.y + sceneViewTransform_.imageSize.y;
199 // SceneView 画像内でも DebugUI パネルが手前にある場合はカーソルを表示する
200 shouldShow = !insideScene || debugUIHovered_;
201 }
202 }
203#endif
204
205 if (shouldShow != appliedCursorVisible_) {
206 ShowCursor(shouldShow ? TRUE : FALSE);
207 appliedCursorVisible_ = shouldShow;
208 }
209}
