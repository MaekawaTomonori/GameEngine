338 void Canvas::Load(const std::string& _name) {
339 std::ifstream file(std::string(JSON_PATH) + _name);
340 if (!file.is_open()) {
341 Log::Send(Log::Level::WARNING, "Canvas: JSON file not found - " + _name);
342 return;
343 }
344
345 nlohmann::json root;
346 try {
347 file >> root;
348 file.close();
349 } catch (const nlohmann::json::parse_error& e) {
350 Log::Send(Log::Level::ERR, "Canvas: JSON parse error - " + std::string(e.what()));
351 file.close();
352 return;
353 }
354
355 if (root.contains("Position") && root["Position"].is_array() && root["Position"].size() >= 2) {
356 position_ = { root["Position"][0].get<float>(), root["Position"][1].get<float>() };
357 } else {
358 position_ = {};
359 }
360
361 elements_.clear();
362
363 if (!root.contains("Elements") || !root["Elements"].is_array()) {
364 ResolveAnimFuncs();
365 return;
366 }
367
368 for (const auto& j : root["Elements"]) {
369 const std::string type = j.value("Type", "Sprite");
370 auto element = CreateElement(type);
371 element->Initialize();
372
373 element->SetName(j.value("Name", "NoName"));
374 element->SetVisible(j.value("Visible", true));
375 element->SetShowAnimKey(j.value("ShowAnim", ""));
376 element->SetIdleAnimKey(j.value("IdleAnim", ""));
377 element->SetHideAnimKey(j.value("HideAnim", ""));
378
379 if (j.contains("Position") && j["Position"].is_array() && j["Position"].size() >= 2) {
380 element->SetPosition({ j["Position"][0].get<float>(), j["Position"][1].get<float>() });
381 }
382 if (j.contains("Color") && j["Color"].is_array() && j["Color"].size() >= 4) {
383 element->SetColor({
384 j["Color"][0].get<float>(), j["Color"][1].get<float>(),
385 j["Color"][2].get<float>(), j["Color"][3].get<float>() });
386 }
387
388 if (j.contains("Events") && j["Events"].is_object()) {
389 for (auto& [keyStr, val] : j["Events"].items()) {
390 element->SetEvent(StringToEventKey(keyStr), val.get<std::string>());
391 }
392 }
393
394 if (type == "Text") {
395 if (auto* te = dynamic_cast<TextElement*>(element.get())) {
396 TextElement::Data td;
397 td.text = j.value("Text", "");
398 td.fontSize = j.value("FontSize", 32.f);
399 te->SetTextData(td);
400 }
401 } else {
402 if (auto* se = dynamic_cast<SpriteElement*>(element.get())) {
403 SpriteElement::Data sd;
404 sd.texture = j.value("Texture", "white_x16.png");
405 if (j.contains("Size") && j["Size"].is_array() && j["Size"].size() >= 2) {
406 sd.size = { j["Size"][0].get<float>(), j["Size"][1].get<float>() };
407 }
408 if (j.contains("TextureLeftTop") && j["TextureLeftTop"].is_array() && j["TextureLeftTop"].size() >= 2) {
409 sd.textureLeftTop = { j["TextureLeftTop"][0].get<float>(), j["TextureLeftTop"][1].get<float>() };
410 }
411 if (j.contains("TextureSize") && j["TextureSize"].is_array() && j["TextureSize"].size() >= 2) {
412 sd.textureSize = { j["TextureSize"][0].get<float>(), j["TextureSize"][1].get<float>() };
413 }
414 se->SetSpriteData(sd);
415 }
416 }
417
418 element->IsOpen() = defaultOpen_;
419 elements_.push_back(std::move(element));
420 }
421
422 ResolveAnimFuncs();
423 if (phase_ == Phase::Idle) {
424 for (auto& e : elements_) e->PlayIdle();
425 }
426
427 Log::Send(Log::Level::INFO,
428 "Canvas: Loaded " + _name + " (" + std::to_string(elements_.size()) + " elements)");
429 }

431 void Canvas::Save() {
432 if (name_.empty()) {
433 Log::Send(Log::Level::WARNING, "Canvas: Cannot save - no name specified");
434 return;
435 }
436
437 nlohmann::json root;
438 root["Position"] = { position_.x, position_.y };
439
440 nlohmann::json elementsArray = nlohmann::json::array();
441
442 for (const auto& element : elements_) {
443 nlohmann::json j;
444
445 j["Type"] = std::string(element->GetType());
446 j["Name"] = element->GetName();
447 j["Position"] = { element->GetPosition().x, element->GetPosition().y };
448 j["Color"] = { element->GetColor().x, element->GetColor().y,
449 element->GetColor().z, element->GetColor().w };
450
451 if (!element->IsVisible()) j["Visible"] = false;
452 if (!element->GetShowAnimKey().empty()) j["ShowAnim"] = element->GetShowAnimKey();
453 if (!element->GetIdleAnimKey().empty()) j["IdleAnim"] = element->GetIdleAnimKey();
454 if (!element->GetHideAnimKey().empty()) j["HideAnim"] = element->GetHideAnimKey();
455
456 const auto& events = element->GetEvents();
457 if (!events.empty()) {
458 nlohmann::json eventsJson;
459 for (const auto& [eventKey, actionKey] : events) {
460 eventsJson[EventKeyToString(eventKey)] = actionKey;
461 }
462 j["Events"] = eventsJson;
463 }
464
465 if (auto* te = dynamic_cast<const TextElement*>(element.get())) {
466 const auto td = te->GetTextData();
467 j["Text"] = td.text;
468 j["FontSize"] = td.fontSize;
469 } else if (auto* se = dynamic_cast<const SpriteElement*>(element.get())) {
470 const auto sd = se->GetSpriteData();
471 j["Texture"] = sd.texture;
472 j["Size"] = { sd.size.x, sd.size.y };
473 if (sd.textureSize.x > 0.f && sd.textureSize.y > 0.f) {
474 j["TextureLeftTop"] = { sd.textureLeftTop.x, sd.textureLeftTop.y };
475 j["TextureSize"] = { sd.textureSize.x, sd.textureSize.y };
476 }
477 }
478
479 elementsArray.push_back(j);
480 }
481
482 root["Elements"] = elementsArray;
483
484 const std::string filePath = std::string(JSON_PATH) + name_ + ".json";
485 std::filesystem::create_directories(JSON_PATH);
486 std::ofstream file(filePath);
487 if (!file.is_open()) {
488 Log::Send(Log::Level::ERR, "Canvas: Failed to open file for saving - " + filePath);
489 return;
490 }
491
492 file << root.dump(4);
493 file.close();
494
495 Log::Send(Log::Level::INFO,
496 "Canvas: Saved " + name_ + ".json (" + std::to_string(elements_.size()) + " elements)");
497 }
