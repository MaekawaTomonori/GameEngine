16void GltfLoader::LoadGltf(const std::string& _directory, const std::string& _name, const ResourceRepository* _repository) {
17 Assimp::Importer importer;
18 std::string path = _directory + _name + "/" + _name + ".gltf";
19 const aiScene* scene = importer.ReadFile(path, aiProcess_FlipWindingOrder | aiProcess_FlipUVs | aiProcess_Triangulate);
20 if (!scene) {
21 Utils::Alert("GltfLoader::LoadGltf: Failed to load file: " + path + "\nError: " + importer.GetErrorString());
22 Log::Send(Log::Level::ERR, "[GLTF Loader] Failed to load " + path + ": " + importer.GetErrorString());
23 return;
24 }
25 if (!scene->HasMeshes()){
26 Utils::Alert("GltfLoader::LoadGltf: No meshes found in file: " + path);
27 return;
28 }
29 Log::Send(Log::Level::INFO, "[GLTF Loader] Loaded " + path + " by Assimp");
30
31 // Model Loader
32 std::unique_ptr<ModelData> model_ = std::make_unique<ModelData>();
33 model_->name = _name;
34 model_->mesh = _name;
35 model_->root = LoadNode(scene->mRootNode);
36 model_->animation = LoadAnimation(scene, _name);
37 model_->skeleton = CreateSkeleton(model_->root);
38
39 //Mesh
40 MeshData mesh_ = LoadMesh(scene, _name, *model_);
41
42 _repository->GetMeshRepository()->Add(_name, mesh_);
43 _repository->GetModelRepository()->Add(_name, std::move(model_));

46Node GltfLoader::LoadNode(const aiNode* _node) {
47 Node node;
48 node.name = _node->mName.C_Str();
49
50 aiVector3D scale, translate;
51 aiQuaternion rotate;
52 _node->mTransformation.Decompose(scale, rotate, translate);
53 node.transform.scale = Vector3{ .x= scale.x, .y= scale.y, .z= scale.z };
54 node.transform.rotate = Quaternion{.x= rotate.x, .y= -rotate.y, .z= -rotate.z, .w= rotate.w};
55 node.transform.translate = Vector3{ .x = -translate.x, .y = translate.y, .z = translate.z };
56 node.local = MathUtils::Matrix::MakeAffineMatrix(node.transform);
57
58 for (unsigned int i = 0; i < _node->mNumChildren; ++i){
59 node.children.push_back(LoadNode(_node->mChildren[i]));
60 }
61
62 return node;
63}

65std::optional<Animation> GltfLoader::LoadAnimation(const aiScene* _scene, const std::string& _name) {
66 Animation result;
67
68 if (_scene->mNumAnimations <= 0) {
69 Log::Send(Log::Level::WARNING, "GltfLoader::LoadAnimation: No animations found in file: " + _name);
70 return std::nullopt;
71 }
72
73 aiAnimation* animation = _scene->mAnimations[0];
74 float tps = (animation->mTicksPerSecond == 0.) ? 1.f : static_cast<float>(animation->mTicksPerSecond);
75
76 result.duration = static_cast<float>(animation->mDuration / tps);
77
78 for (uint32_t channelIndex = 0; channelIndex < animation->mNumChannels; ++channelIndex) {
79 aiNodeAnim* channel = animation->mChannels[channelIndex];
80 NodeAnimation& nodeAnimation = result.nodeAnimations[channel->mNodeName.C_Str()];
81
82 uint32_t key;
83 for (key = 0; key < channel->mNumPositionKeys; ++key) {
84 aiVectorKey& position = channel->mPositionKeys[key];
85 KeyframeVector3 keyframe;
86 keyframe.time = static_cast<float>(position.mTime / tps);
87 keyframe.value = {-position.mValue.x, position.mValue.y, position.mValue.z};
88 nodeAnimation.translate.keyframes.push_back(keyframe);
89 }
90
91 for (key = 0; key < channel->mNumRotationKeys; ++key) {
92 aiQuatKey& rotation = channel->mRotationKeys[key];
93 KeyframeQuaternion keyframe;
94 keyframe.time = static_cast<float>(rotation.mTime / tps);
95 keyframe.value = { rotation.mValue.x, -rotation.mValue.y, -rotation.mValue.z, rotation.mValue.w };
96 nodeAnimation.rotation.keyframes.push_back(keyframe);
97 }
98
99 for (key = 0; key < channel->mNumScalingKeys; ++key) {
100 aiVectorKey& scale = channel->mScalingKeys[key];
101 KeyframeVector3 keyframe;
102 keyframe.time = static_cast<float>(scale.mTime / tps);
103 keyframe.value = { scale.mValue.x, scale.mValue.y, scale.mValue.z };
104 nodeAnimation.scale.keyframes.push_back(keyframe);
105 }
106 }
107
108 return result;

140MeshData GltfLoader::LoadMesh(const aiScene* _scene, const std::string& _name, ModelData& _model) const {
141 MeshData data;
142 for (uint32_t meshIndex = 0; meshIndex < _scene->mNumMeshes; ++meshIndex){
143 aiMesh* mesh = _scene->mMeshes[meshIndex];
144 if (!mesh->HasNormals() || !mesh->HasTextureCoords(0)) {
145 Log::Send(Log::Level::ERR, "GltfLoader::LoadMesh: Mesh does not have normals or texture coordinates");
146 Utils::Alert("Mesh does not have normals or texture coordinates");
147 continue;
148 }
149
150 LoadVertexData(mesh, data);
151
152 LoadIndexData(mesh, data);
153
154 LoadBones(mesh, _model);
155
156 for (uint32_t materialIndex = 0; materialIndex < _scene->mNumMaterials; ++materialIndex){
157 aiMaterial* material = _scene->mMaterials[materialIndex];
158 if (material->GetTextureCount(aiTextureType_DIFFUSE)){
159 aiString texturePath;
160 material->GetTexture(aiTextureType_DIFFUSE, 0, &texturePath);
161 data.texture = ASSETS_FOLDER + _name + "/" + texturePath.C_Str();
162 }
163 }
164 }
165 return data;

198void GltfLoader::LoadBones(const aiMesh* _mesh, ModelData& _model) {
199 for (uint32_t boneIndex = 0; boneIndex < _mesh->mNumBones; ++boneIndex){
200 aiBone* bone = _mesh->mBones[boneIndex];
201 std::string jointName = bone->mName.C_Str();
202 JointWeightData& jointWeight = _model.skinCluster[jointName];
203
204 aiMatrix4x4 bindPose = bone->mOffsetMatrix.Inverse();
205 aiVector3D scale, translate;
206 aiQuaternion rotate;
207
208 bindPose.Decompose(scale, rotate, translate);
209 jointWeight.inverseBindPose = MathUtils::Matrix::MakeAffineMatrix(
210 Vector3{ .x = scale.x, .y = scale.y, .z = scale.z },
211 Quaternion{ .x = rotate.x, .y = -rotate.y, .z = -rotate.z, .w = rotate.w },
212 Vector3{ .x = -translate.x, .y = translate.y, .z = translate.z }
213 ).Inverse();
214
215 for (uint32_t weightIndex = 0; weightIndex < bone->mNumWeights; ++weightIndex){
216 jointWeight.weights.push_back({
217 .weight = bone->mWeights[weightIndex].mWeight,
218 .index = bone->mWeights[weightIndex].mVertexId
219 });
220 }
221 }
