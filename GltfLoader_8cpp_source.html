47Node GltfLoader::LoadNode(const aiNode* _node) {
48 Node node;
49 node.name = _node->mName.C_Str();
50
51 aiVector3D scale, translate;
52 aiQuaternion rotate;
53 _node->mTransformation.Decompose(scale, rotate, translate);
54 node.transform.scale = Vector3{ .x= scale.x, .y= scale.y, .z= scale.z };
55 node.transform.rotate = Quaternion{.x= rotate.x, .y= -rotate.y, .z= -rotate.z, .w= rotate.w};
56 node.transform.translate = Vector3{ .x = -translate.x, .y = translate.y, .z = translate.z };
57 node.local = MathUtils::Matrix::MakeAffineMatrix(node.transform);
58
59 for (unsigned int i = 0; i < _node->mNumChildren; ++i){
60 node.children.push_back(LoadNode(_node->mChildren[i]));
61 }
62
63 return node;

141MeshData GltfLoader::LoadMesh(const aiScene* _scene, const std::string& _name, ModelData& _model) const {
142 MeshData data;
143 for (uint32_t meshIndex = 0; meshIndex < _scene->mNumMeshes; ++meshIndex){
144 aiMesh* mesh = _scene->mMeshes[meshIndex];
145 if (!mesh->HasNormals() || !mesh->HasTextureCoords(0)) {
146 Log::Send(Log::Level::ERR, "GltfLoader::LoadMesh: Mesh does not have normals or texture coordinates");
147 Utils::Alert("Mesh does not have normals or texture coordinates");
148 continue;
149 }
150
151 LoadVertexData(mesh, data);
152
153 LoadIndexData(mesh, data);
154
155 LoadBones(mesh, _model);
156
157 for (uint32_t materialIndex = 0; materialIndex < _scene->mNumMaterials; ++materialIndex){
158 aiMaterial* material = _scene->mMaterials[materialIndex];
159 if (material->GetTextureCount(aiTextureType_DIFFUSE)){
160 aiString texturePath;

162 data.texture = ASSETS_FOLDER + _name + "/" + texturePath.C_Str();
163 }
164 }
165 }

204
205 aiMatrix4x4 bindPose = bone->mOffsetMatrix.Inverse();

213 Vector3{ .x = -translate.x, .y = translate.y, .z = translate.z }
214 ).Inverse();
215
