240bool TextCommon::LoadFont(const std::string& _path) {
241 std::ifstream file(_path, std::ios::binary | std::ios::ate);
242 if (!file.is_open()) {
243 Log::Send(Log::Level::WARNING,
244 "TextCommon::LoadFont: cannot open file: " + _path);
245 return false;
246 }
247 const auto fileSize = static_cast<size_t>(file.tellg());
248 fontData_.resize(fileSize);
249 file.seekg(0);
250 file.read(reinterpret_cast<char*>(fontData_.data()), static_cast<std::streamsize>(fileSize));
251
252 delete static_cast<stbtt_fontinfo*>(fontInfo_);
253 fontInfo_ = new stbtt_fontinfo{};
254 auto* info = static_cast<stbtt_fontinfo*>(fontInfo_);
255
256 if (!stbtt_InitFont(info, fontData_.data(), 0)) {
257 Log::Send(Log::Level::WARNING,
258 "TextCommon::LoadFont: stbtt_InitFont failed: " + _path);
259 return false;
260 }
261
262 const float scale = stbtt_ScaleForPixelHeight(info, kAtlasFontPx);
263 fontScale_ = scale;
264
265 int ascent, descent, lineGap;
266 stbtt_GetFontVMetrics(info, &ascent, &descent, &lineGap);
267 fontAscent_ = static_cast<float>(ascent) * scale;
268
269 std::vector<uint8_t> atlas(kAtlasW * kAtlasH, 0);
270 glyphs_.clear();
271
272 int curX = 0, curY = 0, rowH = 0;
273
274 for (int cp = 32; cp <= 126; ++cp) {
275 const int glyph = stbtt_FindGlyphIndex(info, cp);
276 if (glyph == 0) continue;
277
278 int w, h, xoff, yoff;
279 uint8_t* sdf = stbtt_GetGlyphSDF(
280 info, scale, glyph,
281 kSdfPadding, kOnEdgeValue, kPixelDistScale,
282 &w, &h, &xoff, &yoff);
283
284 if (!sdf) {
285 int advWidth;
286 stbtt_GetGlyphHMetrics(info, glyph, &advWidth, nullptr);
287 glyphs_[static_cast<uint32_t>(cp)] = { 0, 0, 0, 0, 0, 0,
288 static_cast<float>(advWidth) * scale };
289 continue;
290 }
291
292 if (curX + w > kAtlasW) { curX = 0; curY += rowH; rowH = 0; }
293 if (curY + h > kAtlasH) { stbtt_FreeSDF(sdf, nullptr); break; }
294
295 for (int row = 0; row < h; ++row) {
296 std::memcpy(
297 atlas.data() + (curY + row) * kAtlasW + curX,
298 sdf + row * w,
299 static_cast<size_t>(w));
300 }
301
302 int advWidth;
303 stbtt_GetGlyphHMetrics(info, glyph, &advWidth, nullptr);
304
305 glyphs_[static_cast<uint32_t>(cp)] = {
306 curX, curY, w, h, xoff, yoff,
307 static_cast<float>(advWidth) * scale
308 };
309
310 curX += w;
311 rowH = std::max(rowH, h);
312 stbtt_FreeSDF(sdf, nullptr);
313 }
314
315 atlasKey_ = "__text_atlas__" + _path;
316 if (!texture_->LoadFromRawPixels(
317 atlasKey_,
318 atlas.data(), kAtlasW, kAtlasH,
319 DXGI_FORMAT_R8_UNORM))
320 {
321 Log::Send(Log::Level::WARNING,
322 "TextCommon::LoadFont: failed to upload atlas for: " + _path);
323 return false;
324 }
325
326 fontReady_ = true;
327 Log::Send(Log::Level::INFO, "TextCommon::LoadFont: loaded: " + _path);
328 return true;
329}

343void TextCommon::SubmitEntry(const std::string& _text, float _x, float _y, float _fontSize, float _r, float _g, float _b, float _a) {
344 if (!fontReady_ || !mappedInstances_) return;
345
346 const float sizeScale = _fontSize / kAtlasFontPx;
347 const float baseline = _y + fontAscent_ * sizeScale;
348 float penX = _x;
349
350 for (const unsigned char ch : _text) {
351 if (pendingGlyphs_ >= kMaxGlyphs) break;
352
353 auto it = glyphs_.find(static_cast<uint32_t>(ch));
354 if (it == glyphs_.end()) continue;
355
356 const GlyphMetrics& g = it->second;
357
358 if (g.atlasW == 0) {
359 penX += g.advance * sizeScale;
360 continue;
361 }
362
363 GlyphInstance& inst = mappedInstances_[pendingGlyphs_++];
364 inst.posX = penX + static_cast<float>(g.bearingX) * sizeScale;
365 inst.posY = baseline + static_cast<float>(g.bearingY) * sizeScale;
366 inst.sizeX = static_cast<float>(g.atlasW) * sizeScale;
367 inst.sizeY = static_cast<float>(g.atlasH) * sizeScale;
368 inst.uvX = static_cast<float>(g.atlasX) / static_cast<float>(kAtlasW);
369 inst.uvY = static_cast<float>(g.atlasY) / static_cast<float>(kAtlasH);
370 inst.uvW = static_cast<float>(g.atlasW) / static_cast<float>(kAtlasW);
371 inst.uvH = static_cast<float>(g.atlasH) / static_cast<float>(kAtlasH);
372 inst.r = _r; inst.g = _g; inst.b = _b; inst.a = _a;
373
374 penX += g.advance * sizeScale;
375 }
376}
