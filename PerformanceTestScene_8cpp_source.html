54void PerformanceTestScene::UpdateModelSpawning(float _deltaTime) {
55 if (modelSpawnRatePerSecond_ <= 0.f) return;
56
57 const float interval = 1.f / modelSpawnRatePerSecond_;
58 modelSpawnAccumulator_ += _deltaTime;
59
60 int processedThisFrame = 0;
61 while (modelSpawnAccumulator_ >= interval && processedThisFrame < MAX_SPAWN_PER_FRAME) {
62 if (static_cast<int>(models_.size()) < targetModelCount_) {
63 SpawnModel();
64 } else if (static_cast<int>(models_.size()) > targetModelCount_) {
65 DespawnModel();

67 break;
68 }
69 modelSpawnAccumulator_ -= interval;

71 }
