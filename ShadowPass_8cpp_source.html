109 D3D12_VIEWPORT vp{
110 .TopLeftX = 0.0f,
111 .TopLeftY = 0.0f,
112 .Width = static_cast<float>(ShadowCubeMap::RESOLUTION),

114 .MinDepth = 0.0f,

125 XMMATRIX proj = XMMatrixPerspectiveFovLH(XM_PIDIV2, 1.0f, 0.1f, pl->radius);
126
127 struct FaceDesc { XMFLOAT3 dir; XMFLOAT3 up; };
128 static const FaceDesc kFaces[ShadowCubeMap::FACE_COUNT] = {
129 {{ 1, 0, 0}, { 0, 1, 0}},
130 {{-1, 0, 0}, { 0, 1, 0}},
131 {{ 0, 1, 0}, { 0, 0, -1}},
132 {{ 0, -1, 0}, { 0, 0, 1}},
133 {{ 0, 0, 1}, { 0, 1, 0}},
134 {{ 0, 0, -1}, { 0, 1, 0}},
135 };

137 _cubeMap.TransitionToRenderTarget(cmd);
138 pso_->DrawCall();

140 cmd->RSSetScissorRects(1, &scissor);

142 for (UINT face = 0; face < ShadowCubeMap::FACE_COUNT; ++face) {
143 XMVECTOR dir = XMLoadFloat3(&kFaces[face].dir);
144 XMVECTOR up = XMLoadFloat3(&kFaces[face].up);

157 cmd->SetGraphicsRootConstantBufferView(1, shadowCameraResources_[face]->Get()->GetGPUVirtualAddress());
158 _model.ExecuteShadowDraw();
159 }
160
161 _cubeMap.TransitionToShaderResource(cmd);
162}
