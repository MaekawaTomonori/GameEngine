141std::wstring Shader::FindShaderDirectory() {
142 Log::SendWithContext(Log::Level::INFO, "Searching for shader directory", "SHADER_PATH");
143
144 // List of possible shader directory paths to try
145 std::vector<std::wstring> candidatePaths = {
146 L"Assets/Shaders/", // Original relative path
147 L"./Assets/Shaders/", // Explicit current directory
148 L"../Assets/Shaders/", // Parent directory
149 L"../../Assets/Shaders/", // Grandparent directory
150 L"../Engine/Assets/Shaders/", // Relative to Game project
151 L"../../Engine/Assets/Shaders/", // Two levels up from Game output
152 L"Engine/Assets/Shaders/", // Direct Engine path
153 L"GameTemplate/Engine/Assets/Shaders/" // Full relative path from solution
154 };
155
156 for (const auto& candidatePath : candidatePaths) {
157 try {
158 std::string candidatePathStr = Utils::Convert(candidatePath);
159 if (std::filesystem::exists(candidatePath)) {
160 Log::LogFileOperation("DIRECTORY_CHECK", candidatePathStr, true, "Directory exists");
161
162 // Verify it contains shader files
163 bool hasShaderFiles = false;
164 for (const auto& entry : std::filesystem::directory_iterator(candidatePath)) {
165 if (entry.path().extension() == L".hlsl") {
166 hasShaderFiles = true;
167 break;
168 }
169 }
170
171 if (hasShaderFiles) {
172 Log::LogFileOperation("SHADER_DIRECTORY", candidatePathStr, true, "Valid shader directory found with .hlsl files");
173 return candidatePath;
174 } else {
175 Log::LogFileOperation("SHADER_DIRECTORY", candidatePathStr, false, "Directory exists but contains no .hlsl files");
176 }
177 } else {
178 Log::LogFileOperation("DIRECTORY_CHECK", candidatePathStr, false, "Directory does not exist");
179 }
180 } catch (const std::exception& e) {
181 Log::LogFileOperation("DIRECTORY_CHECK", Utils::Convert(candidatePath), false, std::string("Exception: ") + e.what());
182 }
183 }
184
185 // If no valid path found, log error and return default
186 Log::SendWithContext(Log::Level::ERR, "No valid shader directory found! Using default path.", "SHADER_PATH");
187 Log::LogExecutionContext();
188
189 return L"Assets/Shaders/";
190}
