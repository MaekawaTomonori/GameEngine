285IPostEffect* PostEffectChain::Create(const std::string& _type) {
286 // 既存 instance を検索する。
287 for (auto& effectData : effects_) {
288 if (effectData.type == _type) {
289 return effectData.effect.get();
290 }
291 }
292
293 // Factory が未設定なら生成できない。
294 if (!factory_) {
295 Log::Send(Log::Level::ERR, "PostEffectFactory is not set");
296 return nullptr;
297 }
298
299 // Factory から新しい effect を生成する。
300 auto newEffect = factory_->Create(_type);
301 if (!newEffect) {
302 Log::Send(Log::Level::ERR, std::format("Failed to create effect type: {}", _type));
303 return nullptr;
304 }
305
306 // effect をチェーン配下へ組み込む。
307 newEffect->SetUp(adapter_, srv_);
308 const uint32_t slot = rtvHeap_->Allocate();
309 newEffect->SetRTVHandle(rtvHeap_->GetCPUHandle(slot));
310 newEffect->Initialize();
311
312 effects_.emplace_back(EffectData{ std::move(newEffect), _type, slot, true });
313

321void PostEffectChain::ApplyPendingRemovals() {
322 for (const auto& type : pendingRemovals_) {
323 for (auto it = effects_.begin(); it != effects_.end(); ++it) {
324 if (it->type == type) {
325 rtvHeap_->Free(it->rtvSlot);
326
327 // GPU Based Validationは共有SRVヒープ上のディスクリプタが指すリソースを
328 // 常に使用中とみなすため、破棄前に生存中のrenderTexture_へ向け直す
329 srv_->CreateSRVForTexture2D(it->effect->GetSrvIndex(), renderTexture_->Get(), DXGI_FORMAT_R8G8B8A8_UNORM_SRGB, 1);
330
331 effects_.erase(it);
332 break;
333 }
334 }
335 }
336 pendingRemovals_.clear();
337}

357void PostEffectChain::LoadPermanentConfig() {
358 std::string path = PermanentConfigPath();
359 if (!std::filesystem::exists(path)) return;
360
361 std::ifstream file(path);
362 if (!file.is_open()) {
363 Log::Send(Log::Level::ERR, std::format("Failed to open canvas config: {}", path));
364 return;
365 }
366
367 json configJson;
368 file >> configJson;
369 file.close();
370
371 if (!configJson.contains("effects")) return;
372
373 for (const auto& entry : configJson["effects"]) {
374 std::string type = entry.value("type", "");
375 if (type.empty()) continue;
376
377 IPostEffect* effect = Create(type);
378 if (!effect) continue;
379
380 if (entry.contains("baseParameters")) {
381 effect->ApplyParameters(entry["baseParameters"]);
382 }
383 }
384
385 Log::Send(Log::Level::INFO, std::format("Loaded permanent config for '{}'", name_));
386}

388void PostEffectChain::SavePermanentConfig() const {

391

394 entry["type"] = effectData.type;
395 entry["baseParameters"] = effectData.effect->CaptureCurrentParameters();
396 effectsArray.push_back(entry);

398
399 configJson["effects"] = effectsArray;
400
401 std::filesystem::create_directories("./Assets/Data/PostEffect/Canvases");
402
403 std::ofstream file(PermanentConfigPath());
404 if (!file.is_open()) {
405 Log::Send(Log::Level::ERR, std::format("Failed to save canvas config for '{}'", name_));
406 return;
407 }
408
409 file << configJson.dump(4);
410 file.close();
411
