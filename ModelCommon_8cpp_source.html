486void ModelCommon::Draw(Renderer* _renderer) {
487 std::vector<std::function<void()>> staticPostEffectTasks;
488 std::vector<std::function<void()>> staticNoPostEffectTasks;
489 std::vector<std::function<void()>> skinningPostEffectTasks;
490 std::vector<std::function<void()>> skinningNoPostEffectTasks;
491 std::vector<std::function<void()>> staticTransparentPostTasks;
492 std::vector<std::function<void()>> staticTransparentNoPostTasks;
493 std::vector<std::function<void()>> skinningTransparentPostTasks;
494 std::vector<std::function<void()>> skinningTransparentNoPostTasks;
495
496 {
497 std::lock_guard<std::mutex> lock(mutex_);
498
499 for (const auto& command : staticDrawCommands_) {
500 if (command.applyPostEffects) {
501 staticPostEffectTasks.push_back(command.func);
502 } else {
503 staticNoPostEffectTasks.push_back(command.func);
504 }
505 }
506 staticDrawCommands_.clear();
507
508 for (const auto& command : skinningDrawCommands_) {
509 if (command.applyPostEffects) {
510 skinningPostEffectTasks.push_back(command.func);
511 } else {
512 skinningNoPostEffectTasks.push_back(command.func);
513 }
514 }
515 skinningDrawCommands_.clear();
516
517 for (const auto& command : staticTransparentCommands_) {
518 if (command.applyPostEffects) {
519 staticTransparentPostTasks.push_back(command.func);
520 } else {
521 staticTransparentNoPostTasks.push_back(command.func);
522 }
523 }
524 staticTransparentCommands_.clear();
525
526 for (const auto& command : skinningTransparentCommands_) {
527 if (command.applyPostEffects) {
528 skinningTransparentPostTasks.push_back(command.func);
529 } else {
530 skinningTransparentNoPostTasks.push_back(command.func);
531 }
532 }
533 skinningTransparentCommands_.clear();
534 }
535
536 auto bindShadow = [this]() {
537 if (shadowSrvIndex_ != UINT_MAX && srv_) {
538 srv_->SetGraphicsRootDescriptorTable(9, shadowSrvIndex_);
539 adapter_->GetCommandList()->SetGraphicsRootConstantBufferView(10, shadowCbvAddress_);
540 }
541 };
542
543 // --- 不透明パス（深度書き込みあり）---
544 if (!staticNoPostEffectTasks.empty()) {
545 _renderer->Register([this, staticNoPostEffectTasks, bindShadow]() {
546 if (staticPipeline_) {
547 staticPipeline_->DrawCall();
548 }
549 bindShadow();
550 for (auto& task : staticNoPostEffectTasks) {
551 task();
552 }
553 });

557 _renderer->Register([this, staticPostEffectTasks, bindShadow]() {
558 if (staticPipeline_) {
559 staticPipeline_->DrawCall();
560 }
561 bindShadow();
562 for (auto& task : staticPostEffectTasks) {
563 task();
564 }
565 }, true);
566 }
567
568 if (!skinningNoPostEffectTasks.empty()) {
569 _renderer->Register([this, skinningNoPostEffectTasks, bindShadow]() {
570 if (pipeline_) {
571 pipeline_->DrawCall();
572 }
573 bindShadow();
574 for (auto& task : skinningNoPostEffectTasks) {
575 task();
576 }
577 });
578 }
579
580 if (!skinningPostEffectTasks.empty()) {
581 _renderer->Register([this, skinningPostEffectTasks, bindShadow]() {
582 if (pipeline_) {
583 pipeline_->DrawCall();
584 }
585 bindShadow();
586 for (auto& task : skinningPostEffectTasks) {
587 task();
588 }
589 }, true);
590 }
591
592 // --- 半透明パス（深度書き込みなし、不透明パスの後に描画）---
593 if (!staticTransparentNoPostTasks.empty()) {
594 _renderer->Register([this, staticTransparentNoPostTasks, bindShadow]() {
595 if (staticTransparentPipeline_) {
596 staticTransparentPipeline_->DrawCall();
597 }
598 bindShadow();
599 for (auto& task : staticTransparentNoPostTasks) {
600 task();
601 }
602 });

609 }
610 bindShadow();
611 for (auto& task : staticTransparentPostTasks) {
612 task();
613 }

621 }

636 task();
637 }
638 }, true);
639 }
