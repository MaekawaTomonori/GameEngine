309void ModelCommon::Draw(Renderer* _renderer) {
310 std::vector<std::function<void()>> staticPostEffectTasks;
311 std::vector<std::function<void()>> staticNoPostEffectTasks;
312 std::vector<std::function<void()>> skinningPostEffectTasks;
313 std::vector<std::function<void()>> skinningNoPostEffectTasks;
314
315 {
316 std::lock_guard<std::mutex> lock(mutex_);
317
318 for (const auto& command : staticDrawCommands_) {
319 if (command.applyPostEffects) {
320 staticPostEffectTasks.push_back(command.func);
321 } else {
322 staticNoPostEffectTasks.push_back(command.func);
323 }
324 }
325 staticDrawCommands_.clear();
326
327 for (const auto& command : skinningDrawCommands_) {
328 if (command.applyPostEffects) {
329 skinningPostEffectTasks.push_back(command.func);
330 } else {
331 skinningNoPostEffectTasks.push_back(command.func);
332 }
333 }
334 skinningDrawCommands_.clear();
335 }
336
337 if (!staticNoPostEffectTasks.empty()) {
338 _renderer->Register([this, staticNoPostEffectTasks]() {
339 if (staticPipeline_) {
340 staticPipeline_->DrawCall();
341 }
342 for (auto& task : staticNoPostEffectTasks) {
343 task();
344 }
345 });
346 }
347
348 if (!staticPostEffectTasks.empty()) {
349 _renderer->Register([this, staticPostEffectTasks]() {
350 if (staticPipeline_) {
351 staticPipeline_->DrawCall();
352 }
353 for (auto& task : staticPostEffectTasks) {
354 task();
355 }
356 }, true);
357 }
358
359 if (!skinningNoPostEffectTasks.empty()) {
360 _renderer->Register([this, skinningNoPostEffectTasks]() {
361 if (pipeline_) {
362 pipeline_->DrawCall();
363 }
364 for (auto& task : skinningNoPostEffectTasks) {
365 task();
366 }
367 });
368 }
369
370 if (!skinningPostEffectTasks.empty()) {
371 _renderer->Register([this, skinningPostEffectTasks]() {
372 if (pipeline_) {
373 pipeline_->DrawCall();
374 }
375 for (auto& task : skinningPostEffectTasks) {
376 task();
377 }
378 }, true);
379 }
