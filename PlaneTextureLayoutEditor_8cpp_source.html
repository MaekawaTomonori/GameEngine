183 const float availWidth = ImGui::GetContentRegionAvail().x;
184 const float canvasWidth = std::min(availWidth, 480.f);
185 const float canvasHeight = canvasWidth / aspectRatio_;

187 const ImVec2 origin = ImGui::GetCursorScreenPos();

195 std::vector<int> order(layers_.size());
196 std::iota(order.begin(), order.end(), 0);
197 std::ranges::sort(order, [this](const int a, const int b) { return layers_[a].zOrder < layers_[b].zOrder; });
198
199 for (const int idx : order) {
200 const auto& layer = layers_[idx];
201 const ImVec2 rectMinPx(origin.x + layer.rectMin.x * canvasSize.x, origin.y + layer.rectMin.y * canvasSize.y);
202 const ImVec2 rectMaxPx(origin.x + layer.rectMax.x * canvasSize.x, origin.y + layer.rectMax.y * canvasSize.y);
203 const ImU32 tint = ImGui::ColorConvertFloat4ToU32(ImVec4(layer.color.x, layer.color.y, layer.color.z, layer.color.w));
204
205 if (layer.type == PlaneTextureLayer::Type::Texture) {
206 const uint64_t texId = ResolveImTextureId(layer.textureKey);
207 if (texId != 0) {
208 drawList->AddImage(static_cast<ImTextureID>(texId), rectMinPx, rectMaxPx, ImVec2(0, 0), ImVec2(1, 1), tint);
209 } else {

211 }
212 } else {
213 drawList->AddRectFilled(rectMinPx, rectMaxPx, IM_COL32(50, 50, 70, 160));

215 }
216
217 drawList->AddRect(rectMinPx, rectMaxPx, idx == selectedIndex_ ? IM_COL32(255, 200, 0, 255) : IM_COL32(200, 200, 200, 120));
218 }
219
220 // 手前(zOrder降順)からヒットテストし、ドラッグ移動・リサイズを処理する
221 for (auto it = order.rbegin(); it != order.rend(); ++it) {
222 const int idx = *it;
223 auto& layer = layers_[idx];
224 const ImVec2 rectMinPx(origin.x + layer.rectMin.x * canvasSize.x, origin.y + layer.rectMin.y * canvasSize.y);
225 const ImVec2 rectMaxPx(origin.x + layer.rectMax.x * canvasSize.x, origin.y + layer.rectMax.y * canvasSize.y);

236 const ImVec2 delta = ImGui::GetIO().MouseDelta;
237 const Vector2 size = layer.rectMax - layer.rectMin;
238 layer.rectMin += Vector2{ delta.x / canvasSize.x, delta.y / canvasSize.y };
239 ClampMovedRect(layer, size);
240 }
241
242 // リサイズハンドル（右下角）
243 constexpr float kHandle = 8.f;
244 ImGui::SetCursorScreenPos(ImVec2(rectMaxPx.x - kHandle * 0.5f, rectMaxPx.y - kHandle * 0.5f));
245 ImGui::InvisibleButton("resize", ImVec2(kHandle, kHandle));
246 if (ImGui::IsItemActive() && ImGui::IsMouseDragging(ImGuiMouseButton_Left)) {
247 selectedIndex_ = idx;
248 const ImVec2 delta = ImGui::GetIO().MouseDelta;
249 layer.rectMax.x = std::clamp(layer.rectMax.x + delta.x / canvasSize.x, layer.rectMin.x + kMinLayerSize, 1.f);
