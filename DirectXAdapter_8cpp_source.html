307void DirectXAdapter::Present() {
308 UINT bbi = swapChain_->GetCurrentBackBufferIndex();
309
310 // PostDraw
311 swapChainResources_[bbi]->ChangeState(cList_.Get(), D3D12_RESOURCE_STATE_PRESENT);
312
313 if (FAILED(cList_->Close())){
314 Utils::Alert("Failed to close command list");
315 return;
316 }
317 ID3D12CommandList* commandLists[] = { cList_.Get() };
318 cQueue_->ExecuteCommandLists(_countof(commandLists), commandLists);
319 swapChain_->Present(1, 0);
320
321 Wait();
322
323 if (FAILED(cAllocator_->Reset())){
324 Utils::Alert("Failed to reset command allocator");
325 return;
326 }
327 if (FAILED(cList_->Reset(cAllocator_.Get(), nullptr))){
328 Utils::Alert("Failed to reset command list");

346bool DirectXAdapter::CreateDXGI() {
347 Log::Send(Log::Level::INFO, "Create DXGI");
348
349 //Factory
350 if (FAILED(CreateDXGIFactory(IID_PPV_ARGS(&factory_)))){
351 return false;
352 }
353
354 Log::Send(Log::Level::INFO, "Factory Created");
355
356 //Adaptor
357 for (UINT i = 0; SUCCEEDED(factory_->EnumAdapterByGpuPreference(i, DXGI_GPU_PREFERENCE_HIGH_PERFORMANCE, IID_PPV_ARGS(&adapter_))); ++i){
358 // Check if the adapter supports Direct3D 12
359 DXGI_ADAPTER_DESC3 desc;
360 if (FAILED(adapter_->GetDesc3(&desc))){
361 adapter_ = nullptr;
362 }
363 if (!(desc.Flags & DXGI_ADAPTER_FLAG3_SOFTWARE)){
364 break;
365 }
366 adapter_ = nullptr;
367 }
368
369 if (adapter_ == nullptr)return false;
370
371 Log::Send(Log::Level::INFO, "Adapter Created");
372
373 //Device
374 D3D_FEATURE_LEVEL featureLevels[] = {
375 D3D_FEATURE_LEVEL_12_2,
376 D3D_FEATURE_LEVEL_12_1,
377 D3D_FEATURE_LEVEL_12_0
378 };
379
380 Log::Send(Log::Level::INFO, "Device Loop Begin");
381 for (size_t i = 0; i < _countof(featureLevels); ++i){
382 const char* featureLevelNames[] = {
383 "12.2",
384 "12.1",
385 "12.0"
386 };
387 if (SUCCEEDED(D3D12CreateDevice(adapter_.Get(), featureLevels[i], IID_PPV_ARGS(&device_)))){
388 Log::Send(Log::Level::INFO, "Device Created");
389 Log::Send(Log::Level::INFO, "Feature Level: " + std::string(featureLevelNames[i]));
390 break;
391 }
392
393 Log::Send(Log::Level::ERR, "Failed to create device with feature level: " + std::string(featureLevelNames[i]));
394 }
395
396 if (device_ == nullptr){
397 Log::Send(Log::Level::ERR, "Failed to create device");
398 return false;
399 }
400
401 Log::Send(Log::Level::INFO, "Complete create DXGI");
402 return true;
403}

488bool DirectXAdapter::CreateRTV() {
489 // RTVヒープは初回のみ作成、リサイズ時は既存のものを再利用
490 if (!rtvHeap_) {
491 rtvHeap_ = std::make_unique<Heap>();
492 if (!rtvHeap_->Create(device_.Get(), D3D12_DESCRIPTOR_HEAP_TYPE_RTV, 2, D3D12_DESCRIPTOR_HEAP_FLAG_NONE)){
493 Log::Send(Log::Level::ERR, "Failed to create RTV Heap");
494 return false;
495 }
496 Log::Send(Log::Level::INFO, "RTV Heap Created");
497 }
498
499 // SwapChainバッファを取得してラップ
500 swapChainResources_.resize(2);
501 for (UINT i = 0; i < 2; ++i){
502 Microsoft::WRL::ComPtr<ID3D12Resource> resource;
503 if (FAILED(swapChain_->GetBuffer(i, IID_PPV_ARGS(&resource)))){
504 Log::Send(Log::Level::ERR, "Failed to get swap chain buffer " + std::to_string(i));
505 return false;
506 }
507
508 // 新しいバッファサイズを確認
509 D3D12_RESOURCE_DESC desc = resource->GetDesc();
510 Log::Send(Log::Level::INFO,
511 "SwapChain buffer " + std::to_string(i) + " size: " +
512 std::to_string(desc.Width) + "x" + std::to_string(desc.Height));
513
514 swapChainResources_[i] = std::make_unique<DX12Resource>();
515 swapChainResources_[i]->Create(resource);
516 swapChainResources_[i]->Get()->SetName(std::wstring(L"SwapChain" + std::to_wstring(i)).c_str());
517 }
518 Log::Send(Log::Level::INFO, "Swap Chain Resources Created");
519
520 // RTVディスクリプタを作成（既存のヒープに上書き）
521 D3D12_RENDER_TARGET_VIEW_DESC rtvDesc = {};
522 rtvDesc.Format = DXGI_FORMAT_R8G8B8A8_UNORM_SRGB;
523 rtvDesc.ViewDimension = D3D12_RTV_DIMENSION_TEXTURE2D;
524
525 rtvHandles_.resize(2);
526 rtvHandles_[0] = rtvHeap_->Get()->GetCPUDescriptorHandleForHeapStart();
527 device_->CreateRenderTargetView(swapChainResources_[0]->Get(), &rtvDesc, rtvHandles_[0]);
528
529 rtvHandles_[1].ptr = rtvHandles_[0].ptr + device_->GetDescriptorHandleIncrementSize(D3D12_DESCRIPTOR_HEAP_TYPE_RTV);
530 device_->CreateRenderTargetView(swapChainResources_[1]->Get(), &rtvDesc, rtvHandles_[1]);
531
532 Log::Send(Log::Level::INFO, "RTVs Created");
533 return true;
534}

618 if (_width == windowSize_.first && _height == windowSize_.second) {
619 Log::Send(Log::Level::WARNING, "Window size _unchanged, skipping resize");
620 return;
621 }
622
623 // 最小サイズ制約
624 _width = std::max(_width, static_cast<size_t>(1));
625 _height = std::max(_height, static_cast<size_t>(1));
626
627 Log::Send(Log::Level::INFO,
628 "*** STARTING RESIZE: " + std::to_string(windowSize_.first) + "x" + std::to_string(windowSize_.second) +
629 " → " + std::to_string(_width) + "x" + std::to_string(_height) + " ***");
630
631 // RAII化：GPU同期とリソース解放を専用メソッドで実行
632 SyncGPU();
633 ReleaseSwapChainResources();
634
635 Log::Send(Log::Level::INFO, "Resizing swap chain...");
636
637 // SwapChainをリサイズ
638 HRESULT hr = swapChain_->ResizeBuffers(
639 2, // バッファ数
640 static_cast<UINT>(_width), // 新しい幅
641 static_cast<UINT>(_height), // 新しい高さ
642 DXGI_FORMAT_R8G8B8A8_UNORM, // フォーマット
643 0 // フラグ
644 );
645
646 if (FAILED(hr)) {
647 Log::Send(Log::Level::ERR,
648 "Failed to resize swap chain buffers - HRESULT: 0x" +
649 std::to_string(static_cast<unsigned long>(hr)));
650
651 // デバイス削除エラーの特別処理
652 if (hr == DXGI_ERROR_DEVICE_REMOVED) {
653 HRESULT reason = device_->GetDeviceRemovedReason();
654 Log::Send(Log::Level::ERR,
655 "Device removed! Reason: 0x" + std::to_string(static_cast<unsigned long>(reason)));
656 }
657
658 Utils::Alert("Failed to resize swap chain");
659 return;
660 }
661
662 Log::Send(Log::Level::INFO, "SwapChain resized successfully");
663
664 // ウィンドウサイズを更新
665 windowSize_ = {_width, _height};
666
667 // RTVを再作成
668 if (!CreateRTV()) {
669 Log::Send(Log::Level::ERR, "Failed to recreate RTV after resize");
670 Utils::Alert("Failed to recreate RTV");
671 return;
672 }
673
674 // 深度ステンシルバッファを再作成
675 if (!CreateDSV()) {
676 Log::Send(Log::Level::ERR, "Failed to recreate DSV after resize");
677 Utils::Alert("Failed to recreate DSV");
678 return;
679 }
680
681 // ビューポートとシザー矩形を更新
682 viewport_.Width = static_cast<float>(_width);
683 viewport_.Height = static_cast<float>(_height);
684 scissorRect_.right = static_cast<LONG>(_width);
685 scissorRect_.bottom = static_cast<LONG>(_height);
686
687 Log::Send(Log::Level::INFO, "Window resize completed successfully");
688}
