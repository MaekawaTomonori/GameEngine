133bool TextureManager::Load(const std::string& _fileName) {
134 std::lock_guard<std::mutex> lock(mutex_);
135
136 //Remove FolderPath
137 std::string name = _fileName;
138 size_t pos = 0;
139 while((pos = name.find(folderPath_, pos)) != std::string::npos){
140 name.erase(pos, folderPath_.length());
141 }
142
143 //Check if texture is already loaded
144 if (textures_.contains(name)){
145 return true;
146 }
147
148 assert(!srv_->IsFull());
149
158
159
160 //Load Texture
161 DirectX::ScratchImage img = LoadTexture(name);
162
163 if (!img.GetImages() || img.GetImageCount() == 0) {
164 Log::Send(Log::Level::ERR, std::format("TextureManager::Load: Failed to load texture: {}", name));
165 return false;
166 }
167
168 Texture texture;
169 texture.metadata = img.GetMetadata();
170 texture.resource = adapter_->CreateTextureResource(img.GetMetadata());
171
172 if (!texture.resource) {
173 Log::Send(Log::Level::ERR, std::format("TextureManager::Load: Failed to create texture resource: {}", name));
174 return false;
175 }
176
177 UploadTextureData(texture.resource.get(), img);
178
179 texture.srvIndex = srv_->Allocate();
180 texture.cpuHandle = srv_->GetCPUHandle(texture.srvIndex);
181 texture.gpuHandle = srv_->GetGPUHandle(texture.srvIndex);
182
183 if (texture.metadata.IsCubemap()) {
184 srv_->CreateSRVForCubeMap(texture.srvIndex, texture.resource->Get(), texture.metadata.format);
185 }else{
186 srv_->CreateSRVForTexture2D(texture.srvIndex, texture.resource->Get(), texture.metadata.format, static_cast<UINT>(texture.metadata.mipLevels));
187 }
188
189 textures_[name] = std::move(texture);
190
191 Log::Send(Log::Level::INFO, std::format("TextureManager::Load: {}", name));
192 return true;
193}

195bool TextureManager::LoadFromRawPixels(const std::string& _name, const uint8_t* _pixels, uint32_t _width, uint32_t _height, DXGI_FORMAT _format) {
196 std::lock_guard<std::mutex> lock(mutex_);
197
198 if (textures_.contains(_name)) {
199 return true;
200 }
201
202 assert(!srv_->IsFull());
203
204 // 2D テクスチャ用メタデータを手動構築（ミップなし）
205 DirectX::TexMetadata meta{};
206 meta.width = _width;
207 meta.height = _height;
208 meta.depth = 1;
209 meta.arraySize = 1;
210 meta.mipLevels = 1;
211 meta.format = _format;
212 meta.dimension = DirectX::TEX_DIMENSION_TEXTURE2D;
213
214 // ScratchImage を確保してピクセルをコピー
215 DirectX::ScratchImage img;
216 HRESULT hr = img.Initialize2D(_format, _width, _height, 1, 1);
217 if (FAILED(hr)) {
218 Log::Send(Log::Level::ERR, std::format("TextureManager::LoadFromRawPixels: ScratchImage init failed: {}", _name));
219 return false;
220 }
221
222 const DirectX::Image* imageSlice = img.GetImage(0, 0, 0);
223 if (!imageSlice) {
224 Log::Send(Log::Level::ERR, std::format("TextureManager::LoadFromRawPixels: internal ScratchImage error: {}", _name));
225 return false;
226 }
227
228 // rowPitch が GPU アライメントで広い場合に対応して行ごとにコピー
229 size_t srcRowPitch = static_cast<size_t>(_width) * DirectX::BitsPerPixel(_format) / 8;
230 for (uint32_t row = 0; row < _height; ++row) {
231 memcpy(imageSlice->pixels + row * imageSlice->rowPitch,
232 _pixels + row * srcRowPitch,
233 srcRowPitch);
234 }
235
236 Texture texture;
237 texture.metadata = img.GetMetadata();
238 texture.resource = adapter_->CreateTextureResource(img.GetMetadata());
239
240 if (!texture.resource) {
241 Log::Send(Log::Level::ERR, std::format("TextureManager::LoadFromRawPixels: CreateTextureResource failed: {}", _name));
242 return false;
243 }
244
245 UploadTextureData(texture.resource.get(), img);
246
247 texture.srvIndex = srv_->Allocate();
248 texture.cpuHandle = srv_->GetCPUHandle(texture.srvIndex);
249 texture.gpuHandle = srv_->GetGPUHandle(texture.srvIndex);
250 srv_->CreateSRVForTexture2D(texture.srvIndex, texture.resource->Get(), _format, 1);
251
252 textures_[_name] = std::move(texture);
253 Log::Send(Log::Level::INFO, std::format("TextureManager::LoadFromRawPixels: {}", _name));
