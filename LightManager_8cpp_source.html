130void LightManager::Load() {
131 auto json = Singleton<JsonParams>::GetInstance();
132 json->Load(path);
133
134 rawDirectionalLights_.clear();
135 rawPointLights_.clear();
136 rawSpotLights_.clear();
137 *lightCount_ = {0,0,0};
138 auto data = json->GetGroups(path);
139 for (auto& itr : data) {
140 auto group = itr.second;
141 switch (magic_enum::enum_value<LightType>(std::get<int32_t>(group.at("type")))){
142 case LightType::Directional:
143 Add(LightType::Directional);
144 rawDirectionalLights_.back()->Set(itr.first, {
145 std::get<Vector4>(group.at("color")),
146 std::get<Vector3>(group.at("direction")),
147 std::get<float>(group.at("intensity"))
148 });
149 break;
150 case LightType::Point:
151 Add(LightType::Point);
152 rawPointLights_.back()->Set(itr.first, {
153 std::get<Vector4>(group.at("color")),
154 std::get<Vector3>(group.at("position")),
155 std::get<float>(group.at("intensity")),
156 std::get<float>(group.at("radius")),
157 std::get<float>(group.at("decay")),
158 {0,0}
159 });
160 break;
161 case LightType::Spot:
162 Add(LightType::Spot);
163 rawSpotLights_.back()->Set(itr.first, {
164 std::get<Vector4>(group.at("color")),
165 std::get<Vector3>(group.at("position")),
166 std::get<float>(group.at("intensity")),
167 std::get<Vector3>(group.at("direction")).Normalize(),
168 std::get<float>(group.at("distance")),
169 std::get<float>(group.at("decay")),
170 std::get<float>(group.at("cosAngle")),
171 std::get<float>(group.at("falloffStart")),
172 0
173 });
174 break;
175 }
176 }

255void LightManager::Add(LightType _type) {
256 std::unique_ptr<RawDirectionalLight> directional;
257 std::unique_ptr<RawPointLight> point;
258 std::unique_ptr<RawSpotLight> spot;
259
260 switch (_type){
261 case LightType::Directional:
262 if (MAX_COUNT.dlCount <= ++lightCount_->dlCount){
263 return;
264 }
265 directional = std::make_unique<RawDirectionalLight>();
266 directional->DefaultSetting();
267 if (ref_.has_value()){
268 directional->SetReference(ref_.value());
269 }
270 rawDirectionalLights_.push_back(std::move(directional));
271 break;
272 case LightType::Point:
273 if (MAX_COUNT.plCount <= ++lightCount_->plCount){
274 return;
275 }
276 point = std::make_unique<RawPointLight>();
277 point->DefaultSetting();
278 if (ref_.has_value()){
279 point->SetReference(ref_.value());
280 }
281 rawPointLights_.push_back(std::move(point));
282 break;
283 case LightType::Spot:
284 if (MAX_COUNT.slCount <= ++lightCount_->slCount){
285 return;
286 }
287 spot = std::make_unique<RawSpotLight>();
288 spot->DefaultSetting();
289 if (ref_.has_value()){
290 spot->SetReference(ref_.value());
291 }
292 rawSpotLights_.push_back(std::move(spot));
293 break;
294 }
