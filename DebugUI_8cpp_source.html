44 // windowStates_ の表示状態を imgui.ini に永続化する
45 {
46 ImGuiSettingsHandler handler{};
47 handler.TypeName = "WindowStates";
48 handler.TypeHash = ImHashStr("WindowStates");
49 handler.UserData = this;
50 handler.ReadOpenFn = [](ImGuiContext*, ImGuiSettingsHandler*, const char*) -> void* {
51 return reinterpret_cast<void*>(1);
52 };
53 handler.ReadLineFn = [](ImGuiContext*, ImGuiSettingsHandler* h, void*, const char* line) {
54 auto* self = static_cast<DebugUI*>(h->UserData);
55 char key[256];
56 int val;
57 if (sscanf_s(line, "%255[^=]=%d", key, static_cast<unsigned>(sizeof(key)), &val) == 2) {
58 self->windowStates_[key] = (val != 0);
59 }
60 };
61 handler.WriteAllFn = [](ImGuiContext*, ImGuiSettingsHandler* h, ImGuiTextBuffer* buf) {
62 const auto* self = static_cast<const DebugUI*>(h->UserData);
63 buf->appendf("[%s][Data]\n", h->TypeName);
64 for (const auto& [key, state] : self->windowStates_) {
65 buf->appendf("%s=%d\n", key.c_str(), state.visible ? 1 : 0);
66 }
67 buf->appendf("\n");
68 };
69 ImGui::AddSettingsHandler(&handler);
70 }
71
72 ImGui_ImplDX12_Init(
73 _adapter->GetDevice(),
74 2,
75 DXGI_FORMAT_R8G8B8A8_UNORM_SRGB,
76 heap_->Get(),

124
125 std::ranges::sort(commands, [](const Command& _a, const Command& _b){
126 return _a.id < _b.id;
127 });
128
129 for (const auto &[id, command] : commands) {
130 if (windowStates_.contains(id) && !windowStates_[id].visible) {

132 }

134 command();

175#ifdef _DEBUG
176 if (!_resource || !adapter_) return 0;

178 // スロット0 = ImGuiフォント、スロット1 = シーンテクスチャ

180
181 D3D12_SHADER_RESOURCE_VIEW_DESC srvDesc = {};

250void DebugUI::MenuBar() {
251#ifdef _DEBUG
252 const ImGuiViewport* vp = ImGui::GetMainViewport();
253 const float menuBarHeight = ImGui::GetFrameHeight();
254 ImGui::SetNextWindowPos(ImVec2(vp->Pos.x, vp->Pos.y + menuBarHeight), ImGuiCond_Always);
255 ImGui::SetNextWindowSizeConstraints(ImVec2(160, 0), ImVec2(300, FLT_MAX));
256 ImGui::Begin("##WindowsPanel", &showWindowsPanel_,
257 ImGuiWindowFlags_NoTitleBar |
258 ImGuiWindowFlags_NoResize |
259 ImGuiWindowFlags_NoMove |
260 ImGuiWindowFlags_AlwaysAutoResize |
261 ImGuiWindowFlags_NoSavedSettings);
262
263 bool belong = false;
264 for (auto& state : windowStates_ | std::views::values) {
265 if (state.group.empty()) continue;
266
267 belong = true;
268 break;
269 }
270
271 if (belong) {
272 std::map<std::string, std::vector<std::pair<const std::string*, WindowState*>>> groups;
273 std::vector<std::pair<const std::string*, WindowState*>> others;
274 for (auto& [label, state] : windowStates_) {
275 if (state.group.empty()) {
276 others.emplace_back(&label, &state);
277 } else {
278 groups[state.group].emplace_back(&label, &state);
279 }
280 }
281
282 for (auto& [groupName, items] : groups) {
283 if (ImGui::CollapsingHeader(groupName.c_str())) {
284 for (auto& [label, state] : items) {
285 ImGui::Checkbox(label->c_str(), &state->visible);
286 }
287 }
288 }
289
290 if (!others.empty()) {
291 if (!groups.empty()) {
292 ImGui::Separator();
293 }
294 if (ImGui::CollapsingHeader("Others")) {
295 for (auto& [label, state] : others) {
296 ImGui::Checkbox(label->c_str(), &state->visible);
297 }
298 }
299 }
300 } else {
301 for (auto& [label, state] : windowStates_) {
302 ImGui::Checkbox(label.c_str(), &state.visible);
303 }
304 }
305
306 const ImVec2 panelPos = ImGui::GetWindowPos();
307 const ImVec2 panelSize = ImGui::GetWindowSize();
308 ImGui::End();
309
310 // パネル外クリックで閉じる
311 // 開いた直後のフレームはボタン位置がパネル外になるためスキップ
312 if (panelJustOpened_) {
313 panelJustOpened_ = false;
314 }
315 else if (ImGui::IsMouseReleased(ImGuiMouseButton_Left)) {
316 const ImVec2 m = ImGui::GetMousePos();
317 const bool inPanel = m.x >= panelPos.x && m.x <= panelPos.x + panelSize.x
318 && m.y >= panelPos.y && m.y <= panelPos.y + panelSize.y;
319 if (!inPanel) {
320 showWindowsPanel_ = false;
321 }
322 }
323#endif
324}
